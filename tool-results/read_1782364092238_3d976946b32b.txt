    20→function StarRow({ rating, size = 'w-3.5 h-3.5' }: { rating: number; size?: string }) {

    27→        />

    30→  );
    31→}

    33→function timeAgo(iso: string): string {
    34→  try {
    35→    const then = new Date(iso).getTime();
    36→    const diff = Date.now() - then;
    37→    const mins = Math.floor(diff / 60000);
    38→    if (mins < 1) return 'just now';
    39→    if (mins < 60) return `${mins}m ago`;
    40→    const hours = Math.floor(mins / 60);
    41→    if (hours < 24) return `${hours}h ago`;
    42→    const days = Math.floor(hours / 24);
    43→    if (days < 30) return `${days}d ago`;
    44→    return new Date(iso).toLocaleDateString('en-NG', { month: 'short', day: 'numeric' });
    45→  } catch {
    46→    return '';
    47→  }
    48→}
