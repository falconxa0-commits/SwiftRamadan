    20→const progressSteps = [
    21→  { key: 'confirmed', label: 'Confirmed', threshold: 10 },
    22→  { key: 'preparing', label: 'Preparing', threshold: 35 },
    23→  { key: 'ready', label: 'Ready', threshold: 55 },
    24→  { key: 'transit', label: 'In Transit', threshold: 75 },
    25→  { key: 'delivered', label: 'Delivered', threshold: 100 },
    26→];
    27→
    28→function OrderProgressTracker({ progress, status }: { progress: number; status: string }) {
    29→  return (
    30→    <div className="mt-3">
    31→      <div className="flex items-center justify-between">
    32→        {progressSteps.map((step, i) => {
    33→          const isActive = progress >= step.threshold;
    34→          const isCurrentStep = status === 'Preparing' && step.key === 'preparing' ||
    35→            status === 'In Transit' && step.key === 'transit' ||
    36→            status === 'Delivered' && step.key === 'delivered' ||
    37→            status === 'Confirmed' && step.key === 'confirmed' ||
    38→            status === 'Ready' && step.key === 'ready';
    39→          return (
    40→            <div key={step.key} className="flex items-center flex-1">
    41→              <div className="flex flex-col items-center">
    42→                <div className={`w-5 h-5 rounded-full flex items-center justify-center border-2 transition-colors ${
    43→                  isActive
    44→                    ? isCurrentStep
    45→                      ? 'border-[#10E07A] bg-[#10E07A]'
    46→                      : 'border-[#10E07A]/50 bg-[#10E07A]/20'
    47→                    : 'border-white/10 bg-[#0F1118]'
    48→                }`}>
    49→                  {isActive && (
    50→                    <div className={`w-1.5 h-1.5 rounded-full ${isCurrentStep ? 'bg-[#06070B]' : 'bg-[#10E07A]'}`} />
    51→                  )}
    52→                </div>
    53→                <span className={`text-[8px] mt-1 font-bold whitespace-nowrap ${
    54→                  isActive ? 'text-[#10E07A]' : 'text-white/20'
    55→                }`}>{step.label}</span>
    56→              </div>
    57→              {i < progressSteps.length - 1 && (
    58→                <div className={`flex-1 h-px mx-1 ${progress >= progressSteps[i + 1].threshold ? 'bg-[#10E07A]/40' : 'bg-white/5'}`} />
    59→              )}
    60→            </div>
    61→          );
    62→        })}
    63→      </div>
    64→    </div>
    65→  );
    66→}
