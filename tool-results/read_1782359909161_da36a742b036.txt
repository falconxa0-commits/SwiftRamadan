    10→interface CountdownTime {
    11→  hours: number;
    12→  minutes: number;
    13→  seconds: number;
    14→}
    15→
    16→function useCountdown(endTime: Date): CountdownTime {
    17→  const calculate = () => {
    18→    const now = new Date();
    19→    const diff = endTime.getTime() - now.getTime();
    20→    if (diff <= 0) return { hours: 0, minutes: 0, seconds: 0 };
    21→    return {
    22→      hours: Math.floor(diff / (1000 * 60 * 60)),
    23→      minutes: Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)),
    24→      seconds: Math.floor((diff % (1000 * 60)) / 1000),
    25→    };
    26→  };
    27→
    28→  const [timeLeft, setTimeLeft] = useState<CountdownTime>(calculate);
    29→
    30→  useEffect(() => {
    31→    const interval = setInterval(() => {
    32→      setTimeLeft(calculate());
    33→    }, 1000);
    34→
    35→    return () => clearInterval(interval);
    36→  }, [endTime]);
    37→
    38→  return timeLeft;
    39→}
