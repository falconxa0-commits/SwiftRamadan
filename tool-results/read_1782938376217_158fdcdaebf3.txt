   101→
   102→function ProgressBar({ step, total, accent }: { step: number; total: number; accent: string }) {
   103→  return (
   104→    <div className="flex items-center gap-2 w-full">
   105→      {Array.from({ length: total }).map((_, i) => (
   106→        <div key={i} className="flex-1 flex flex-col items-center gap-1.5">
   107→          <div
   108→            className="w-full h-1.5 rounded-full transition-all duration-500"
   109→            style={{
   110→              backgroundColor: i <= step ? accent : 'rgba(255,255,255,0.1)',
   111→              boxShadow: i === step ? `0 0 8px ${accent}40` : 'none',
   112→            }}
   113→          />
   114→        </div>
   115→      ))}
   116→      <span className="text-white/30 text-xs font-medium ml-2 shrink-0">
   117→        Step {step + 1} of {total}
   118→      </span>
   119→    </div>
   120→  );
   121→}

   123→/* ──────────────────── Confetti Particle ──────────────────── */
   124→
   125→function ConfettiParticle({ delay, color }: { delay: number; color: string }) {
   126→  const randomX = Math.random() * 300 - 150;
   127→  const randomY = Math.random() * 400 + 100;
   128→  const randomRotate = Math.random() * 720 - 360;
   129→  const size = Math.random() * 8 + 4;
   130→
   131→  return (
   132→    <motion.div
   133→      initial={{ opacity: 1, x: 0, y: 0, rotate: 0, scale: 1 }}
   134→      animate={{ opacity: 0, x: randomX, y: randomY, rotate: randomRotate, scale: 0.3 }}
   135→      transition={{ duration: 1.8, delay, ease: 'easeOut' }}
   136→      className="absolute rounded-sm"
   137→      style={{
   138→        width: size,
   139→        height: size,
   140→        backgroundColor: color,
   141→        left: '50%',
   142→        top: '40%',
   143→      }}
   144→    />
   145→  );
   146→}

  1096→   MAIN ONBOARDING FLOW
  1097→   ════════════════════════════════════════════════════════════════ */
  1098→
  1099→export default function OnboardingFlow() {
  1100→  const {
  1101→    showOnboarding,
  1102→    setShowOnboarding,
  1103→    onboardingStep,
  1104→    setOnboardingStep,
  1105→    onboardingComplete,
  1106→    setOnboardingComplete,
  1107→    userRole,
  1108→    setActiveTab,
  1109→    setIsLoggedIn,
  1110→  } = useAppStore();
  1111→
  1112→  const [showCelebration, setShowCelebration] = useState(false);
  1113→
  1114→  const role = userRole as 'customer' | 'vendor' | 'rider';
  1115→  const accent = ROLE_ACCENT[role] || ROLE_ACCENT.customer;
  1116→  const btnClass = ROLE_BTN_CLASS[role] || ROLE_BTN_CLASS.customer;
  1117→  const ctaText = ROLE_CTA[role] || ROLE_CTA.customer;
  1118→  const totalSteps = 3;
  1119→
  1120→  const getStepContent = () => {
  1121→    switch (userRole) {
  1122→      case 'customer':
  1123→        return [
  1124→          <CustomerStep1 key="c1" />,
  1125→          <CustomerStep2 key="c2" />,
  1126→          <CustomerStep3 key="c3" />,
  1127→        ];
  1128→      case 'vendor':
  1129→        return [
  1130→          <VendorStep1 key="v1" />,
  1131→          <VendorStep2 key="v2" />,
  1132→          <VendorStep3 key="v3" />,
  1133→        ];
  1134→      case 'rider':
  1135→        return [
  1136→          <RiderStep1 key="r1" />,
  1137→          <RiderStep2 key="r2" />,
  1138→          <RiderStep3 key="r3" />,
  1139→        ];
  1140→      default:
  1141→        return [<CustomerStep1 key="c1" />, <CustomerStep2 key="c2" />, <CustomerStep3 key="c3" />];
  1142→    }
  1143→  };
  1144→
  1145→  const steps = getStepContent();
  1146→
  1147→  const handleNext = () => {
  1148→    if (onboardingStep < totalSteps - 1) {
  1149→      setOnboardingStep(onboardingStep + 1);
  1150→    } else {
  1151→      // Complete onboarding - show celebration
  1152→      setShowCelebration(true);
  1153→    }
  1154→  };
  1155→
  1156→  const handleBack = () => {
  1157→    if (onboardingStep > 0) {
  1158→      setOnboardingStep(onboardingStep - 1);
  1159→    }
  1160→  };
  1161→
  1162→  const handleSkip = () => {
  1163→    // Per Task 3d: Skip completes onboarding immediately (no celebration)
  1164→    // and navigates the user to the main app.
  1165→    setOnboardingComplete(true);
  1166→    setIsLoggedIn(true);
  1167→    setShowOnboarding(false);
  1168→    setActiveTab(ROLE_DEFAULT_TAB[role]);
  1169→    setOnboardingStep(0);
  1170→    toast({
  1171→      title: 'Onboarding skipped ⏭️',
  1172→      description: `Welcome to SwiftRamadan! You can complete your profile later in Settings.`,
  1173→    });
  1174→  };
  1175→
  1176→  const handleCelebrationDone = () => {
  1177→    setOnboardingComplete(true);
  1178→    setIsLoggedIn(true);
  1179→    setShowOnboarding(false);
  1180→    setActiveTab(ROLE_DEFAULT_TAB[role]);
  1181→    setOnboardingStep(0);
  1182→    setShowCelebration(false);
  1183→    toast({
  1184→      title: `Welcome to SwiftRamadan! 🌙`,
  1185→      description: `Your ${role} experience is ready.`,
  1186→    });
  1187→  };
  1188→
  1189→  // Reset celebration state when onboarding reopens
  1190→  const handleOnboardingClose = () => {
  1191→    if (!showCelebration) {
  1192→      setShowCelebration(true);
  1193→    }
  1194→  };
  1195→
  1196→  if (!showOnboarding || onboardingComplete) return null;
  1197→
  1198→  // Show celebration overlay
  1199→  if (showCelebration) {
  1200→    return (
  1201→      <AnimatePresence>
  1202→        <CelebrationScreen role={role} onDone={handleCelebrationDone} />
  1203→      </AnimatePresence>
  1204→    );
  1205→  }
  1206→
  1207→  return (
  1208→    <AnimatePresence>
  1209→      <motion.div
  1210→        initial={{ opacity: 0 }}
  1211→        animate={{ opacity: 1 }}
  1212→        exit={{ opacity: 0 }}
  1213→        transition={{ duration: 0.25 }}
  1214→        className="fixed inset-0 z-[120] bg-[#05070A] flex flex-col"
  1215→      >
  1216→        {/* Top Bar */}
  1217→        <div className="flex items-center justify-between px-4 h-14 shrink-0 border-b border-white/5">
  1218→          <div className="flex items-center gap-3">
  1219→            {onboardingStep > 0 ? (
  1220→              <button
  1221→                onClick={handleBack}
  1222→                className="w-10 h-10 rounded-full bg-[#1A1D26] border border-white/10 flex items-center justify-center hover:border-white/20 transition-colors"
  1223→                aria-label="Go back"
  1224→              >
  1225→                <ArrowLeft className="w-5 h-5 text-white" />
  1226→              </button>
  1227→            ) : (
  1228→              <div className="w-10" />
  1229→            )}
  1230→          </div>
  1231→
  1232→          {/* Progress */}
  1233→          <div className="flex-1 mx-4">
  1234→            <ProgressBar step={onboardingStep} total={totalSteps} accent={accent} />
  1235→          </div>
  1236→
  1237→          {/* Skip Button */}
  1238→          <button
  1239→            onClick={handleSkip}
  1240→            className="text-white/40 text-sm font-semibold hover:text-white/60 transition-colors px-2"
  1241→          >
  1242→            Skip
  1243→          </button>
  1244→        </div>
  1245→
  1246→        {/* Step Content */}
  1247→        <div className="flex-1 overflow-y-auto custom-scrollbar">
  1248→          <AnimatePresence mode="wait">
  1249→            {steps[onboardingStep]}
  1250→          </AnimatePresence>
  1251→        </div>
  1252→
  1253→        {/* Bottom Action Button */}
  1254→        <div className="shrink-0 px-6 py-5 border-t border-white/5 bg-[#05070A]">
  1255→          <motion.button
  1256→            onClick={handleNext}
  1257→            whileTap={{ scale: 0.97 }}
  1258→            className={`w-full h-14 rounded-xl font-bold text-base shadow-lg active:scale-[0.98] transition-transform flex items-center justify-center gap-2 ${btnClass}`}
  1259→          >
  1260→            {onboardingStep === totalSteps - 1 ? (
  1261→              <>
  1262→                <Sparkles className="w-5 h-5" />
  1263→                {ctaText}
  1264→              </>
  1265→            ) : (
  1266→              <>
  1267→                Continue
  1268→                <ChevronDown className="w-4 h-4 rotate-[-90deg]" />
  1269→              </>
  1270→            )}
  1271→          </motion.button>
  1272→        </div>
  1273→
  1274→        {/* Bottom Safe Area */}
  1275→        <div className="shrink-0 h-2 bg-[#05070A]" />
  1276→      </motion.div>
  1277→    </AnimatePresence>
  1278→  );
  1279→}
