    34→type TimeOfDay = 'sahur' | 'morning' | 'afternoon' | 'iftar' | 'evening' | 'night';
    35→
    36→function getTimeOfDay(): TimeOfDay {
    37→  const hour = new Date().getHours();
    38→  if (hour >= 3 && hour < 5) return 'sahur';
    39→  if (hour >= 5 && hour < 12) return 'morning';
    40→  if (hour >= 12 && hour < 15) return 'afternoon';
    41→  if (hour >= 15 && hour < 19) return 'iftar';
    42→  if (hour >= 19 && hour < 21) return 'evening';
    43→  return 'night';
    44→}

   107→/* ──────────────────── Simple Markdown Renderer ──────────────────── */
   108→
   109→function renderBotText(text: string): React.ReactNode[] {
   110→  const lines = text.split('\n');
   111→  const result: React.ReactNode[] = [];
   112→
   113→  for (let i = 0; i < lines.length; i++) {
   114→    const line = lines[i];
   115→
   116→    // Bullet points
   117→    if (line.match(/^[-•]\s/)) {
   118→      const content = line.replace(/^[-•]\s/, '');
   119→      result.push(
   120→        <div key={i} className="flex gap-2 ml-1">
   121→          <span className="text-[#F5C451] shrink-0">•</span>
   122→          <span>{renderInlineFormatting(content)}</span>
   123→        </div>
   124→      );
   125→      continue;
   126→    }
   127→
   128→    // Numbered lists
   129→    if (line.match(/^\d+\.\s/)) {
   130→      const content = line.replace(/^\d+\.\s/, '');
   131→      const num = line.match(/^(\d+)/)?.[1];
   132→      result.push(
   133→        <div key={i} className="flex gap-2 ml-1">
   134→          <span className="text-[#10E07A] shrink-0 font-semibold">{num}.</span>
   135→          <span>{renderInlineFormatting(content)}</span>
   136→        </div>
   137→      );
   138→      continue;
   139→    }
   140→
   141→    // Empty lines
   142→    if (line.trim() === '') {
   143→      result.push(<div key={i} className="h-2" />);
   144→      continue;
   145→    }
   146→
   147→    // Regular lines
   148→    result.push(<span key={i}>{renderInlineFormatting(line)}</span>);
   149→  }
   150→
   151→  return result;
   152→}
   153→
   154→function renderInlineFormatting(text: string): React.ReactNode[] {
   155→  const parts: React.ReactNode[] = [];
   156→  // Match **bold** and *italic*
   157→  const regex = /(\*\*(.+?)\*\*|\*(.+?)\*)/g;
   158→  let lastIndex = 0;
   159→  let match: RegExpExecArray | null;
   160→  let key = 0;
   161→
   162→  while ((match = regex.exec(text)) !== null) {
   163→    // Add text before match
   164→    if (match.index > lastIndex) {
   165→      parts.push(<span key={key++}>{text.slice(lastIndex, match.index)}</span>);
   166→    }
   167→
   168→    if (match[2]) {
   169→      // Bold
   170→      parts.push(<strong key={key++} className="text-[#F5C451] font-semibold">{match[2]}</strong>);
   171→    } else if (match[3]) {
   172→      // Italic
   173→      parts.push(<em key={key++}>{match[3]}</em>);
   174→    }
   175→
   176→    lastIndex = match.index + match[0].length;
   177→  }
   178→
   179→  // Add remaining text
   180→  if (lastIndex < text.length) {
   181→    parts.push(<span key={key++}>{text.slice(lastIndex)}</span>);
   182→  }
   183→
   184→  return parts.length > 0 ? parts : [text];
   185→}

   187→/* ──────────────────── LocalStorage Persistence ──────────────────── */
   188→
   189→const CHAT_STORAGE_KEY = 'safa-chat-history';
   190→const MAX_STORED_MESSAGES = 50;
   191→
   192→function loadMessages(): Message[] {
   193→  if (typeof window === 'undefined') return [];
   194→  try {
   195→    const stored = localStorage.getItem(CHAT_STORAGE_KEY);
   196→    if (stored) {
   197→      const parsed = JSON.parse(stored);
   198→      if (Array.isArray(parsed) && parsed.length > 0) return parsed;
   199→    }
   200→  } catch { /* ignore */ }
   201→  return [];
   202→}
   203→
   204→function saveMessages(messages: Message[]) {
   205→  if (typeof window === 'undefined') return;
   206→  try {
   207→    localStorage.setItem(CHAT_STORAGE_KEY, JSON.stringify(messages.slice(-MAX_STORED_MESSAGES)));
   208→  } catch { /* ignore */ }
   209→}

   211→/* ──────────────────── Format Time ──────────────────── */
   212→
   213→function formatTime(timestamp: number): string {
   214→  const d = new Date(timestamp);
   215→  return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
   216→}
