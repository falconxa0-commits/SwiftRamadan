    80→/* ──────────────────── Live Countdown Component ──────────────────── */
    81→
    82→function IftarCountdown({ minutesUntilIftar }: { minutesUntilIftar: number }) {
    83→  const [secondsLeft, setSecondsLeft] = useState(minutesUntilIftar * 60);
    84→
    85→  useEffect(() => {
    86→    const interval = setInterval(() => {
    87→      setSecondsLeft((prev) => {
    88→        if (prev <= 1) {
    89→          clearInterval(interval);
    90→          return 0;
    91→        }
    92→        return prev - 1;
    93→      });
    94→    }, 1000);
    95→    return () => clearInterval(interval);
    96→  }, [minutesUntilIftar]);
    97→
    98→  const mins = Math.floor(secondsLeft / 60);
    99→  const secs = secondsLeft % 60;
   100→  const isUrgent = mins <= 15;
   101→
   102→  return (
   103→    <div
   104→      className={`flex items-center gap-1.5 px-2.5 py-1 rounded-full backdrop-blur-sm border ${
   105→        isUrgent ? 'bg-red-500/90 border-red-400/30 animate-pulse' : 'bg-red-500/90 border-red-400/30'
   106→      }`}
   107→    >
   108→      <Timer className="w-3 h-3 text-white" />
   109→      <span className="text-white text-[10px] font-black font-mono">
   110→        {mins}:{secs.toString().padStart(2, '0')} to Iftar
   111→      </span>
   112→    </div>
   113→  );
   114→}

   274→        throw new Error(json.error || 'Failed to accept order');
   275→      }

   277→      console.error(err);
   278→      toast({
   279→        title: 'Accept failed',
   280→        description: 'Could not accept order. Please try again.',
   281→        variant: 'destructive',
   282→      });

   284→      setProcessing((p) => {
   285→        const next = { ...p };

   287→        return next;

   305→          title: 'Order Rejected ❌',

   353→      });
