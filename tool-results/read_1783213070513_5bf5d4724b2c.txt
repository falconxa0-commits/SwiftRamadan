   239→  const [loading, setLoading] = useState(false);
   240→
   241→  const config = ROLE_CONFIG[loginRole];
   242→
   243→  const handleLogin = async () => {
   244→    if (!email.trim() || !password.trim()) {
   245→      toast({ title: 'Missing fields', description: 'Please enter your email and password.', variant: 'destructive' });
   246→      return;
   247→    }
   248→    setLoading(true);
   249→    try {
   250→      const res = await fetch('/api/auth', {
   251→        method: 'POST',
   252→        headers: { 'Content-Type': 'application/json' },
   253→        body: JSON.stringify({ action: 'login', email, password, role: loginRole }),
   254→      });
   255→      const data = await res.json();
   256→      if (data.success) {
   257→        setUserName(data.user?.name || email.split('@')[0]);
   258→        setUserEmail(email);
   259→        setUserRole(loginRole);
   260→        setIsLoggedIn(true);
   261→        setShowAuth(null);
   262→        track('login', { role: loginRole, method: 'password' });
   263→        toast({ title: `Welcome back!`, description: `Signed in as ${config.label}` });
   264→      } else {
   265→        // Demo mode: allow login even without DB account for seamless experience
   266→        setUserName(email.split('@')[0]);
   267→        setUserEmail(email);
   268→        setUserRole(loginRole);
   269→        setIsLoggedIn(true);
   270→        setShowAuth(null);
   271→        track('login', { role: loginRole, method: 'demo' });
   272→        toast({ title: `Welcome back!`, description: `Signed in as ${config.label} (demo)` });

   274→    } catch {
   275→      // Fallback: demo login

   489→
   490→  // Total steps depend on role
   491→  const totalSteps = signupRole === 'customer' ? 2 : 3;
   492→  const currentStep = step === 1 ? 1 : (signupRole === 'customer' ? 2 : 2);
   493→
   494→  const handleStep1Next = () => {
   495→    if (!fullName.trim() || !phone.trim() || !email.trim() || !area) {
   496→      toast({ title: 'Missing fields', description: 'Please fill in all required fields.', variant: 'destructive' });
   497→      return;
   498→    }
   499→    // If customer, go directly to OTP (skip step 2)
   500→    if (signupRole === 'customer') {
   501→      handleSubmit();
   502→    } else {
   503→      setStep(2);
   504→    }
   505→  };
   506→
   507→  const handleSubmit = async () => {
   508→    setLoading(true);
   509→
   510→    // Set role-specific fields in store
   511→    store.setUserRole(signupRole);
   512→
   513→    if (signupRole === 'vendor') {
   514→      store.setVendorStoreName(businessName);
   515→      store.setVendorBusinessCategory(businessCategory);
   516→      store.setVendorBusinessAddress(businessAddress);

   519→      store.setRiderPlateNumber(plateNumber);
   520→      store.setRiderLicenseNumber(licenseNumber);

   524→      const res = await fetch('/api/auth', {
   525→        method: 'POST',
   526→        headers: { 'Content-Type': 'application/json' },
   527→        body: JSON.stringify({
   528→          action: 'signup',
   529→          name: fullName,
   530→          phone: `+234${phone}`,
   531→          email,

   972→
   973→function OTPScreen() {
   974→  const { userPhone, userEmail, userRole, setUserRole, setShowAuth, setIsLoggedIn, setShowOnboarding } = useAppStore();
   975→  const [otp, setOtp] = useState<string[]>(Array(6).fill(''));
   976→  const [countdown, setCountdown] = useState(60);
   977→  const [canResend, setCanResend] = useState(false);
   978→  const [loading, setLoading] = useState(false);
   979→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
   980→
   981→  const config = ROLE_CONFIG[userRole || 'customer'];
   982→  const accentColor = userRole ? config.accent : '#13ec13';
   983→
   984→  useEffect(() => {
   985→    if (countdown <= 0) {
   986→      setCanResend(true);
   987→      return;
   988→    }
   989→    const timer = setInterval(() => {
   990→      setCountdown((prev) => prev - 1);
   991→    }, 1000);
   992→    return () => clearInterval(timer);
   993→  }, [countdown]);
   994→
   995→  const handleChange = useCallback((index: number, value: string) => {
   996→    if (!/^\d*$/.test(value)) return;
   997→    const newOtp = [...otp];
   998→    newOtp[index] = value.slice(-1);
   999→    setOtp(newOtp);
  1000→    if (value && index < 5) {
  1001→      inputRefs.current[index + 1]?.focus();
  1002→    }
  1003→  }, [otp]);
  1004→
  1005→  const handleKeyDown = useCallback((index: number, e: React.KeyboardEvent<HTMLInputElement>) => {
  1006→    if (e.key === 'Backspace' && !otp[index] && index > 0) {
  1007→      inputRefs.current[index - 1]?.focus();
  1008→    }
  1009→  }, [otp]);
  1010→
  1011→  const handlePaste = useCallback((e: React.ClipboardEvent) => {
  1012→    e.preventDefault();
  1013→    const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6);
  1014→    if (pasted.length > 0) {
  1015→      const newOtp = [...otp];
  1016→      for (let i = 0; i < pasted.length; i++) {
  1017→        newOtp[i] = pasted[i];
  1018→      }
  1019→      setOtp(newOtp);
  1020→      const nextEmpty = pasted.length < 6 ? pasted.length : 5;
  1021→      inputRefs.current[nextEmpty]?.focus();
  1022→    }
  1023→  }, [otp]);
  1024→
  1025→  const handleVerifySuccess = () => {
  1026→    setIsLoggedIn(true);
  1027→    track('signup', { role: userRole || 'customer' });
  1028→    if (userRole && userRole !== 'customer') {
  1029→      setShowOnboarding(true);
  1030→      setShowAuth(null);
  1031→      toast({ title: 'Verified! 🎉', description: `Setting up your ${ROLE_CONFIG[userRole].label} account...` });
  1032→    } else if (userRole === 'customer') {
  1033→      setShowOnboarding(true);
  1034→      setShowAuth(null);
  1035→      toast({ title: 'Verified! 🎉', description: 'Welcome to SwiftRamadan!' });
  1036→    } else {
  1037→      setShowAuth('role');
  1038→      toast({ title: 'Verified! 🎉', description: 'Please choose how you\'d like to use SwiftRamadan.' });
  1039→    }
  1040→  };
  1041→
  1042→  const handleVerify = async () => {
  1043→    const code = otp.join('');
  1044→    if (code.length < 6) {
