    64→/* ───────── Helpers ───────── */
    65→
    66→function itemsSummary(items: OrderItem[]): string {
    67→  if (!items || items.length === 0) return 'No items';
    68→  return items.map((i) => `${i.qty}x ${i.name}`).join(', ');
    69→}
    70→
    71→function shortId(id: string): string {
    72→  return id.slice(-6).toUpperCase();
    73→}
    74→
    75→function timeAgo(iso: string): string {
    76→  const diff = Date.now() - new Date(iso).getTime();
    77→  const mins = Math.floor(diff / 60000);
    78→  if (mins < 1) return 'just now';
    79→  if (mins < 60) return `${mins}m ago`;
    80→  const hrs = Math.floor(mins / 60);
    81→  if (hrs < 24) return `${hrs}h ago`;
    82→  const days = Math.floor(hrs / 24);
    83→  return `${days}d ago`;
    84→}
