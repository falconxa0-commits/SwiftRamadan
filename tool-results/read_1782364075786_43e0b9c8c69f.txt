    95→   Helpers

    97→
    98→function initials(name: string): string {
    99→  const parts = name.trim().split(/\s+/);
   100→  if (parts.length === 0) return '?';
   101→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   102→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   103→}
   104→
   105→function formatTime(ts: number): string {
   106→  const d = new Date(ts);
   107→  return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
   108→}
   109→
   110→/** Map an Order.status (DB string) → internal DeliveryStatus */
   111→function mapOrderStatus(status: string): DeliveryStatus {
   112→  switch (status) {
   113→    case 'Confirmed':
   114→    case 'Preparing':
   115→      return 'preparing';
   116→    case 'Ready':
   117→      return 'picked_up';
   118→    case 'In Transit':
   119→      return 'on_the_way';
   120→    case 'Delivered':
   121→      return 'delivered';
   122→    default:
   123→      return 'preparing';
   124→  }
   125→}
   126→
   127→/** Map Order.progress (0-100) → ETA minutes (inverse) */
   128→function progressToEta(progress: number, status: DeliveryStatus): number {
   129→  if (status === 'delivered') return 0;
   130→  if (status === 'arriving') return 2;
   131→  if (status === 'on_the_way') return Math.max(3, Math.round((100 - progress) / 6));
   132→  if (status === 'picked_up') return Math.max(5, Math.round((100 - progress) / 6));
   133→  return Math.max(8, Math.round((100 - progress) / 6));
   134→}
   135→
   136→/** Generate system messages based on status */
   137→function statusMessage(status: DeliveryStatus, riderName: string): string {
   138→  switch (status) {
   139→    case 'preparing':
   140→      return 'Your order is being prepared with care 🍳';
   141→    case 'picked_up':
   142→      return `${riderName} has picked up your order 🏍️`;
   143→    case 'on_the_way':
   144→      return `${riderName} is on the way to you 📍`;
   145→    case 'arriving':
   146→      return `${riderName} is arriving at your location 🚪`;
   147→    case 'delivered':
   148→      return `Delivered! Ramadan Mubarak 🌙 — Enjoy your meal!`;
   149→    default:
   150→      return 'Updating...';
   151→  }
   152→}

   188→      if (!res.ok) return;

   214→        name: riderName,

   237→        rider,
   238→        location: { lat: locLat, lng: locLng },

   268→      });

   276→    }

   385→              <div className="flex items-center gap-3 min-w-0">

   402→                  <p className="text-white/40 text-[11px] truncate">
   403→                    {delivery

   407→                </div>

   409→              <button
   410→                onClick={handleClose}
