    16→/* ───────── Types ───────── */
    17→
    18→interface OrderItem {
    19→  name: string;
    20→  qty: number;
    21→  price: number;
    22→}
    23→
    24→interface RiderOrder {
    25→  id: string;
    26→  status: string;
    27→  total: number;
    28→  riderName: string | null;
    29→  items: OrderItem[];
    30→  progress: number;
    31→  createdAt: string;
    32→}
    33→
    34→interface WeeklyEarning {
    35→  day: string;
    36→  amount: number;
    37→}
    38→
    39→interface RiderData {
    40→  riderName: string;
    41→  online: boolean;
    42→  rating: number;
    43→  completedToday: number;
    44→  earningsToday: number;
    45→  totalEarnings: number;

    53→
    54→/* ───────── Animation variants ───────── */
    55→
    56→const staggerContainer = {
    57→  hidden: { opacity: 0 },

    59→};
    60→

    65→
    66→/* ───────── Helpers ───────── */
    67→
    68→function itemsSummary(items: OrderItem[]): string {
    69→  if (!items || items.length === 0) return 'No items';
    70→  return items.map((i) => `${i.qty}x ${i.name}`).join(', ');
    71→}
    72→
    73→function shortId(id: string): string {
    74→  return id.slice(-6).toUpperCase();
    75→}
    76→
    77→function timeAgo(iso: string): string {
    78→  const diff = Date.now() - new Date(iso).getTime();
    79→  const mins = Math.floor(diff / 60000);
    80→  if (mins < 1) return 'just now';
    81→  if (mins < 60) return `${mins}m ago`;
    82→  const hrs = Math.floor(mins / 60);
    83→  if (hrs < 24) return `${hrs}h ago`;
    84→  const days = Math.floor(hrs / 24);
    85→  return `${days}d ago`;
    86→}

   126→      osc.frequency.exponentialRampToValueAtTime(990, ctx.currentTime + 0.18); // B5
   127→      gain.gain.setValueAtTime(0.0001, ctx.currentTime);

   198→        /* store not ready */
   199→      }

   379→            >
   380→              {riderOnline ? 'Online' : 'Offline'}
   381→            </p>
   382→            <p className="text-white/30 text-[10px] flex items-center gap-1">
   383→              {socketConnected ? (
   384→                <>
   385→                  <BellRing className="w-3 h-3 text-[#10E07A]" />

   387→                  <span>· listening for requests</span>
   388→                </>
   389→              ) : (
   390→                'Toggle to receive deliveries'

   392→            </p>
   393→          </div>
   394→        </div>
   395→        <div
   396→          className={`flex items-center gap-2 px-3 py-1.5 rounded-full text-xs font-bold ${
   397→            riderOnline
   398→              ? 'bg-[#38BDF8]/10 text-[#38BDF8] border border-[#38BDF8]/20'
