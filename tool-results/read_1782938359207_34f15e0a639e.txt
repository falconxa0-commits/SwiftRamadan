   102→/* ──────────────────── Live Countdown Component ──────────────────── */
   103→
   104→function IftarCountdown({ minutesUntilIftar }: { minutesUntilIftar: number }) {
   105→  const [secondsLeft, setSecondsLeft] = useState(minutesUntilIftar * 60);
   106→
   107→  useEffect(() => {
   108→    const interval = setInterval(() => {
   109→      setSecondsLeft((prev) => {
   110→        if (prev <= 1) {
   111→          clearInterval(interval);
   112→          return 0;
   113→        }
   114→        return prev - 1;
   115→      });
   116→    }, 1000);
   117→    return () => clearInterval(interval);
   118→  }, [minutesUntilIftar]);
   119→
   120→  const mins = Math.floor(secondsLeft / 60);
   121→  const secs = secondsLeft % 60;
   122→  const isUrgent = mins <= 15;
   123→
   124→  return (
   125→    <div
   126→      className={`flex items-center gap-1.5 px-2.5 py-1 rounded-full backdrop-blur-sm border ${
   127→        isUrgent ? 'bg-red-500/90 border-red-400/30 animate-pulse' : 'bg-red-500/90 border-red-400/30'
   128→      }`}
   129→    >
   130→      <Timer className="w-3 h-3 text-white" />
   131→      <span className="text-white text-[10px] font-black font-mono">
   132→        {mins}:{secs.toString().padStart(2, '0')} to Iftar
   133→      </span>
   134→    </div>
   135→  );
   136→}

   225→export default function VendorDashboard() {
   226→  const {

   241→  const [loading, setLoading] = useState(true);
   242→  const [processing, setProcessing] = useState<Record<string, 'accept' | 'reject' | 'ready' | undefined>>({});
   243→  // Local list of incoming order ids that the vendor has rejected/accepted
   244→  const [hiddenIds, setHiddenIds] = useState<Set<string>>(new Set());
   245→  // Marked-as-ready processing orders (for quick visual feedback)
   246→  const [readyIds, setReadyIds] = useState<Set<string>>(new Set());
   247→  // All vendor orders (polled from /api/vendor/orders) — used to derive the
   248→  // Processing & Dispatched tabs. The /api/vendor endpoint only returns the
   249→  // "incoming" slice, so we keep this second list for accepted/dispatched orders.
   250→  const [allOrders, setAllOrders] = useState<VendorApiOrder[]>([]);
   251→
   252→  // ─── Realtime: join vendor room when we have the vendorId ───
   253→  // We use the vendor's email as a stable identifier when the API
   254→  // doesn't expose a numeric id (vendorId may be null on first load).
   255→  const vendorRoomId = data?.vendorId
   256→    ? `vendor-${data.vendorId}`
   257→    : userEmail
   258→      ? `vendor-${userEmail}`
   259→      : undefined;
   260→  const { socket, isConnected: socketConnected } = useSocket(vendorRoomId);
   261→
   262→  /** Play a short notification chime using the Web Audio API. */
   263→  const playChime = useCallback(() => {
   264→    try {
   265→      const AudioCtx =
   266→        (window as unknown as { AudioContext?: typeof AudioContext; webkitAudioContext?: typeof AudioContext })
   267→          .AudioContext ||
   268→        (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
   269→      if (!AudioCtx) return;
   270→      const ctx = new AudioCtx();
   271→      const osc = ctx.createOscillator();
   272→      const gain = ctx.createGain();
   273→      osc.type = 'sine';
   274→      osc.frequency.setValueAtTime(880, ctx.currentTime); // A5
   275→      osc.frequency.exponentialRampToValueAtTime(1320, ctx.currentTime + 0.12); // E6
   276→      gain.gain.setValueAtTime(0.0001, ctx.currentTime);
   277→      gain.gain.exponentialRampToValueAtTime(0.15, ctx.currentTime + 0.02);
   278→      gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.4);
   279→      osc.connect(gain);
   280→      gain.connect(ctx.destination);
   281→      osc.start();
   282→      osc.stop(ctx.currentTime + 0.42);
   283→      // Close context shortly after to free resources
   284→      setTimeout(() => ctx.close().catch(() => {}), 600);
   285→    } catch {
   286→      /* Audio API not available */
   287→    }
   288→  }, []);
   289→
   290→  // Listen for new-order events
   291→  useEffect(() => {
   292→    if (!socket) return;
   293→
   294→    const onNewOrder = (payload: {
   295→      vendorId?: string;
   296→      orderData?: Record<string, unknown> & {
   297→        id?: string;
   298→        customer?: string;
   299→        area?: string;
   300→        total?: number;
   301→        items?: { name: string; qty: number; price: number }[];
   302→        image?: string;
   303→        minutesUntilIftar?: number;
   304→      };
   305→      timestamp?: string;
   306→    }) => {
   307→      if (!payload) return;
   308→      const od = payload.orderData || {};
   309→      const newOrder: IncomingOrder = {
   310→        id:
   311→          (typeof od.id === 'string' && od.id) ||
   312→          `SWR-${Date.now().toString(36).toUpperCase()}`,
   313→        customer: (typeof od.customer === 'string' && od.customer) || 'New customer',
   314→        area: (typeof od.area === 'string' && od.area) || 'Lagos',
   315→        items: Array.isArray(od.items) ? od.items : [],
   316→        total: typeof od.total === 'number' ? od.total : 0,
   317→        minutesUntilIftar:
   318→          typeof od.minutesUntilIftar === 'number' ? od.minutesUntilIftar : 30,
   319→        status: 'incoming',
   320→        image:
   321→          (typeof od.image === 'string' && od.image) ||
   322→          '/images/meals/meal-jollof.png',
   323→        createdAt: new Date().toISOString(),
   324→        progress: 0,
   325→      };
   326→
   327→      // Prepend to incoming orders if not already present
   328→      setData((prev) => {
   329→        if (!prev) return prev;
   330→        if (prev.incomingOrders.some((o) => o.id === newOrder.id)) return prev;
   331→        return {
   332→          ...prev,
   333→          incomingOrders: [newOrder, ...prev.incomingOrders],
   334→          todayOrders: prev.todayOrders + 1,
   335→        };
   336→      });
   337→
   338→      // Remove from hiddenIds in case it was hidden before
   339→      setHiddenIds((s) => {
   340→        const next = new Set(s);
   341→        next.delete(newOrder.id);
   342→        return next;
   343→      });
   344→
   345→      // Switch to "incoming" tab so the vendor sees it immediately
   346→      setActiveFilter('incoming');
   347→
   348→      // Toast + chime
   349→      toast({
   350→        title: 'New order received! 🔔',
   351→        description: `${newOrder.customer} • ${formatNaira(newOrder.total)}`,
   352→      });
   353→      playChime();
   354→    };
   355→
   356→    socket.on('new-order', onNewOrder);
   357→    return () => {
   358→      socket.off('new-order', onNewOrder);
   359→    };
   360→  }, [socket, toast, playChime]);
   361→
   362→  /* ── Fetch vendor dashboard data ── */
   363→  const fetchData = useCallback(async () => {
   364→    setLoading(true);
   365→    try {
   366→      const res = await fetch(`/api/vendor?email=${encodeURIComponent(userEmail || 'sani@swiftramadan.app')}`);
   367→      const json = await res.json();
   368→      if (json.success && json.data) {
   369→        setData(json.data);
   370→        if (json.data.storeName) setVendorStoreName(json.data.storeName);
   371→        setVendorBalance(json.data.balance);
   372→        setVendorPendingSettlement(json.data.pendingSettlement);
   373→        setVendorTotalEarnings(json.data.totalEarnings);
   374→      }
   375→    } catch (err) {
   376→      console.error('[VendorDashboard] fetch error:', err);
   377→      toast({
   378→        title: 'Failed to load dashboard',
   379→        description: 'Please check your connection and try again.',
   380→        variant: 'destructive',
   381→      });
   382→    } finally {
   383→      setLoading(false);
   384→    }
   385→  }, [userEmail, setVendorStoreName, setVendorBalance, setVendorPendingSettlement, setVendorTotalEarnings, toast]);
   386→
   387→  useEffect(() => {
   388→    fetchData();
   389→  }, [fetchData]);
   390→
   391→  /* ── Fetch all vendor orders (for Processing & Dispatched tabs) ── */
   392→  const fetchVendorOrders = useCallback(async () => {
   393→    try {
   394→      const res = await fetch(
   395→        `/api/vendor/orders?email=${encodeURIComponent(userEmail || 'sani@swiftramadan.app')}`,
   396→      );
   397→      const json = await res.json();
   398→      if (json.success && Array.isArray(json.orders)) {
   399→        setAllOrders(json.orders);
   400→      }
   401→    } catch (err) {
   402→      console.error('[VendorDashboard] fetch orders error:', err);
   403→    }
   404→  }, [userEmail]);
   405→
   406→  useEffect(() => {
   407→    fetchVendorOrders();
   408→  }, [fetchVendorOrders]);
   409→
   410→  /* ── Accept order ── */
   411→  const handleAccept = async (order: IncomingOrder) => {
   412→    setProcessing((p) => ({ ...p, [order.id]: 'accept' }));
   413→    try {
   414→      const res = await fetch('/api/vendor/orders', {
   415→        method: 'PUT',
   416→        headers: { 'Content-Type': 'application/json' },

   423→          title: 'Order Accepted! ✅',

   430→          fetchData();
   431→          fetchVendorOrders();
   432→        }, 600);
   433→      } else {
   434→        throw new Error(json.error || 'Failed to accept order');
   435→      }
   436→    } catch (err) {
   437→      console.error(err);
   438→      toast({
   439→        title: 'Accept failed',
   440→        description: 'Could not accept order. Please try again.',
   441→        variant: 'destructive',
   442→      });
   443→    } finally {
   444→      setProcessing((p) => {
   445→        const next = { ...p };
   446→        delete next[order.id];
   447→        return next;
   448→      });
   449→    }
   450→  };
   451→

   476→        throw new Error(json.error || 'Failed to reject order');
   477→      }
   478→    } catch (err) {
   479→      console.error(err);
   480→      toast({
   481→        title: 'Reject failed',
   482→        description: 'Could not reject order. Please try again.',
   483→        variant: 'destructive',
   484→      });
   485→    } finally {
   486→      setProcessing((p) => {
   487→        const next = { ...p };
   488→        delete next[order.id];
   489→        return next;
   490→      });
   491→    }
   492→  };

   494→  /* ── Mark ready (for processing orders) ── */

   507→          title: 'Order Ready! 🎉',
   508→          description: `Order ${orderId.slice(-6).toUpperCase()} marked as ready for dispatch`,

   511→        setTimeout(() => {
   512→          fetchData();
   513→          fetchVendorOrders();
   514→        }, 600);
   515→      } else {
   516→        throw new Error(json.error || 'Failed to mark order ready');
   517→      }
   518→    } catch (err) {
   519→      console.error(err);
   520→      toast({
   521→        title: 'Action failed',
   522→        description: 'Could not mark order as ready.',
   523→        variant: 'destructive',
   524→      });
   525→    } finally {
   526→      setProcessing((p) => {
   527→        const next = { ...p };
   528→        delete next[orderId];
   529→        return next;
   530→      });
   531→    }
   532→  };
   533→
   534→  /* ── Toggle online status ── */
   535→  const handleToggleOnline = async () => {
   536→    const next = !vendorOnline;
   537→    setVendorOnline(next);
   538→    toast({
   539→      title: next ? 'Back Online! 🟢' : 'Going Offline',
   540→      description: next
   541→        ? 'You are now accepting orders for Iftar & Suhoor'
   542→        : 'You will stop receiving new orders',
   543→    });
   544→    try {
   545→      await fetch('/api/vendor', {
   546→        method: 'POST',
   547→        headers: { 'Content-Type': 'application/json' },
   548→        body: JSON.stringify({ action: 'toggle-online', email: userEmail, online: next }),
   549→      });
   550→    } catch (err) {
   551→      console.error('Toggle online failed:', err);
   552→    }
   553→  };
   554→
   555→  /* ── Derived lists ── */
   556→  // /api/vendor returns incomingOrders for BOTH 'Preparing' (truly incoming)
   557→  // and 'Confirmed' (already accepted) statuses. To prevent accepted orders
   558→  // from showing in BOTH the Incoming and Processing tabs (especially after a
   559→  // page refresh, when hiddenIds is empty), we exclude any order whose id is
   560→  // present in allOrders with a status that belongs to Processing or Dispatched.
   561→  const otherTabOrderIds = new Set(
   562→    allOrders
   563→      .filter(
   564→        (o) =>
   565→          o.status === 'Confirmed' ||
   566→          o.status === 'Ready' ||
   567→          o.status === 'In Transit',
   568→      )
   569→      .map((o) => o.id),
   570→  );
   571→  const incomingOrders = (data?.incomingOrders || []).filter(
   572→    (o) => !hiddenIds.has(o.id) && !otherTabOrderIds.has(o.id),
   573→  );
   574→
   575→  // Processing: vendor has accepted the order (status 'Confirmed') and is preparing it.
