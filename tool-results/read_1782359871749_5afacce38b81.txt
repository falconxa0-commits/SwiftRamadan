    21→function ConfettiParticle({ delay, color }: { delay: number; color: string }) {
    22→  const randomX = Math.random() * 300 - 150;
    23→  const randomY = Math.random() * 400 + 100;
    24→  const randomRotate = Math.random() * 720 - 360;
    25→  const size = Math.random() * 8 + 4;
    26→
    27→  return (
    28→    <motion.div
    29→      initial={{ opacity: 1, x: 0, y: 0, rotate: 0, scale: 1 }}
    30→      animate={{ opacity: 0, x: randomX, y: randomY, rotate: randomRotate, scale: 0.3 }}
    31→      transition={{ duration: 1.8, delay, ease: 'easeOut' }}
    32→      className="absolute rounded-sm pointer-events-none"
    33→      style={{
    34→        width: size,
    35→        height: size,
    36→        backgroundColor: color,
    37→        left: '50%',
    38→        top: '30%',
    39→      }}
    40→    />
    41→  );
    42→}
