    34→function timeAgo(iso: string): string {
    35→  try {
    36→    const then = new Date(iso).getTime();
    37→    const diff = Date.now() - then;
    38→    const mins = Math.floor(diff / 60000);
    39→    if (mins < 1) return 'just now';
    40→    if (mins < 60) return `${mins}m ago`;
    41→    const hours = Math.floor(mins / 60);
    42→    if (hours < 24) return `${hours}h ago`;
    43→    const days = Math.floor(hours / 24);
    44→    if (days < 30) return `${days}d ago`;
    45→    return new Date(iso).toLocaleDateString('en-NG', { month: 'short', day: 'numeric' });
    46→  } catch {
    47→    return '';
    48→  }
    49→}
