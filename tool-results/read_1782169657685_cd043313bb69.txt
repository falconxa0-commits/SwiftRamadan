   140→/* ─────────────── Hero Banner ─────────────── */
   141→function HeroBanner() {
   142→  const [currentSlide, setCurrentSlide] = useState(0);
   143→
   144→  useEffect(() => {
   145→    const timer = setInterval(() => {
   146→      setCurrentSlide(prev => (prev + 1) % heroSlides.length);
   147→    }, 4000);
   148→    return () => clearInterval(timer);
   149→  }, []);
   150→
   151→  return (
   152→    <div className="relative mx-4 mt-3">
   153→      <AnimatePresence mode="wait">
   154→        <motion.div
   155→          key={currentSlide}
   156→          initial={{ opacity: 0, scale: 1.02 }}
   157→          animate={{ opacity: 1, scale: 1 }}
   158→          exit={{ opacity: 0, scale: 0.98 }}
   159→          transition={{ duration: 0.5 }}
   160→          className="relative w-full aspect-[2/1] rounded-2xl overflow-hidden"
   161→        >
   162→          <div
   163→            className="absolute inset-0 bg-center bg-no-repeat bg-cover"
   164→            style={{ backgroundImage: `url("${heroSlides[currentSlide].image}")` }}
   165→          />
   166→          <div className="absolute inset-0" style={{
   167→            background: 'linear-gradient(135deg, rgba(8,11,18,0.85) 0%, rgba(8,11,18,0.4) 50%, rgba(8,11,18,0.7) 100%)',
   168→          }} />
   169→
   170→          {heroSlides[currentSlide].badge && (
   171→            <div className="absolute top-3 left-3 flex items-center gap-1 px-2.5 py-1 rounded-full"
   172→              style={{ background: 'linear-gradient(135deg, #D4AF37, #F5E6A3)', boxShadow: '0 2px 12px rgba(212,175,55,0.3)' }}>
   173→              <Star className="w-3 h-3 text-[#080B12] fill-[#080B12]" />
   174→              <span className="text-[#080B12] text-[9px] font-black uppercase tracking-wider">{heroSlides[currentSlide].badge}</span>
   175→            </div>
   176→          )}
   177→
   178→          <div className="absolute bottom-0 left-0 right-0 p-4">
   179→            <h2 className="text-white text-lg font-extrabold leading-tight tracking-tight">
   180→              {heroSlides[currentSlide].title}
   181→            </h2>
   182→            <div className="flex items-center gap-1.5 mt-1">
   183→              <Zap className="w-3 h-3 text-[#13ec13]" />
   184→              <span className="text-[#13ec13]/80 text-xs font-semibold">{heroSlides[currentSlide].subtitle}</span>
   185→            </div>
   186→          </div>

   189→
   190→      <div className="flex justify-center gap-1.5 mt-2.5">
   191→        {heroSlides.map((_, i) => (
   192→          <button
   193→            key={i}
   194→            onClick={() => setCurrentSlide(i)}

   196→              i === currentSlide ? 'w-6 bg-[#D4AF37]' : 'w-1.5 bg-white/15'
   197→            }`}
   198→          />
   199→        ))}
   200→      </div>
   201→    </div>
   202→  );
   203→}
