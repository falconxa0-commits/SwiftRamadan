     4→import { motion, AnimatePresence } from 'framer-motion';

     6→import {
     7→  MapPin,
     8→  Bike,
     9→  Phone,
    10→  MessageSquare,

    12→  Star,
    13→  Check,
    14→  Navigation,
    15→  X,
    16→  Clock,

    18→  Store,
    19→  User,
    20→} from 'lucide-react';

    23→
    24→/* ──────────────────────────────────────────────────────────
    25→   Types
    26→   ────────────────────────────────────────────────────────── */
    27→
    28→type DeliveryStatus =
    29→  | 'preparing'
    30→  | 'picked_up'
    31→  | 'on_the_way'
    32→  | 'arriving'
    33→  | 'delivered';
    34→
    35→interface GeoPoint {
    36→  lat: number;
    37→  lng: number;
    38→}
    39→
    40→interface Rider {
    41→  name: string;
    42→  phone: string;
    43→  photo: string;
    44→  rating: number;
    45→  vehicle: string;
    46→  color: string;
    47→}
    48→
    49→interface DeliveryState {
    50→  orderId: string;
    51→  rider: Rider;
    52→  location: GeoPoint;
    53→  status: DeliveryStatus;
    54→  eta: number;
    55→  progress: number;
    56→  customer: GeoPoint;
    57→  store: { name: string; lat: number; lng: number };
    58→}
    59→
    60→interface ChatMessage {
    61→  orderId: string;
    62→  from: 'rider' | 'customer' | 'system';
    63→  text: string;
    64→  timestamp: number;
    65→}
    66→
    67→/* ──────────────────────────────────────────────────────────
    68→   Status config
    69→   ────────────────────────────────────────────────────────── */
    70→
    71→const STATUS_STAGES: Array<{
    72→  key: DeliveryStatus;
    73→  label: string;
    74→  desc: string;
    75→  icon: typeof Package;
    76→}> = [
    77→  { key: 'preparing', label: 'Order Placed', desc: 'Restaurant preparing your meal', icon: Package },
    78→  { key: 'picked_up', label: 'Picked Up', desc: 'Rider collected your order', icon: Store },
    79→  { key: 'on_the_way', label: 'On The Way', desc: 'Heading to your location', icon: Bike },
    80→  { key: 'arriving', label: 'Arriving', desc: 'Rider is at your doorstep', icon: Navigation },
    81→  { key: 'delivered', label: 'Delivered', desc: 'Order completed - Ramadan Mubarak!', icon: Check },
    82→];
    83→
    84→const STATUS_ORDER: DeliveryStatus[] = [
    85→  'preparing',
    86→  'picked_up',
    87→  'on_the_way',
    88→  'arriving',
    89→  'delivered',
    90→];
    91→
    92→/* ──────────────────────────────────────────────────────────
    93→   Helpers
    94→   ────────────────────────────────────────────────────────── */
    95→
    96→function initials(name: string): string {
    97→  const parts = name.trim().split(/\s+/);
    98→  if (parts.length === 0) return '?';
    99→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   100→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   101→}
   102→
   103→function formatTime(ts: number): string {
   104→  const d = new Date(ts);
   105→  return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
   106→}

   108→/* ──────────────────────────────────────────────────────────
   109→   Component
   110→   ────────────────────────────────────────────────────────── */
   111→
   112→export default function RealTimeTrackingModal() {
   113→  const { activeModal, setActiveModal, orders } = useAppStore();
   114→  const { toast } = useToast();
   115→  const isOpen = activeModal === 'live-tracking';
   116→
   117→  const socketRef = useRef<Socket | null>(null);
   118→  const [isConnected, setIsConnected] = useState(false);
   119→
   120→  const [delivery, setDelivery] = useState<DeliveryState | null>(null);
   121→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   122→  const [chatInput, setChatInput] = useState('');
   123→
   124→  const [activeOrderId, setActiveOrderId] = useState<string>(
   125→    () => useAppStore.getState().orders?.[0]?.id ?? 'SWR-2847'
   126→  );
   127→
   128→  const messagesEndRef = useRef<HTMLDivElement>(null);
   129→  const chatScrollRef = useRef<HTMLDivElement>(null);
   130→
   131→  // Keep latest activeOrderId available to socket handlers without re-running effect
   132→  const activeOrderIdRef = useRef(activeOrderId);
   133→  useEffect(() => {
   134→    activeOrderIdRef.current = activeOrderId;
   135→  }, [activeOrderId]);
   136→
   137→  /* Connect to socket.io when modal opens */
   138→  useEffect(() => {
   139→    if (!isOpen) return;
   140→
   141→    // CRITICAL: must use relative path with XTransformPort - never use absolute URL
   142→    const sock = io('/?XTransformPort=3003', {
   143→      transports: ['websocket', 'polling'],
   144→      forceNew: true,
   145→      reconnection: true,
   146→      reconnectionAttempts: 5,
   147→      reconnectionDelay: 1000,
   148→      timeout: 10000,
   149→    });
   150→
   151→    socketRef.current = sock;
   152→
   153→    sock.on('connect', () => {
   154→      setIsConnected(true);
   155→    });
   156→
   157→    sock.on('disconnect', () => {
   158→      setIsConnected(false);
   159→    });
   160→
   161→    // Server sends snapshot of all active deliveries on connect
   162→    sock.on('active_deliveries', (list: DeliveryState[]) => {
   163→      if (!list || list.length === 0) return;
   164→      // Prefer the user's selected order, else pick the first
   165→      const match = list.find((d) => d.orderId === activeOrderIdRef.current);
   166→      const chosen = match || list[0];
   167→      setDelivery(chosen);
   168→      setActiveOrderId(chosen.orderId);
   169→      sock.emit('subscribe_order', chosen.orderId);
   170→    });
   171→
   172→    sock.on('location_update', (update: DeliveryState) => {
   173→      if (!update) return;
   174→      setDelivery((prev) => {
   175→        // If we have no delivery yet, adopt this one if it's our active order
   176→        if (!prev) {
   177→          if (update.orderId === activeOrderIdRef.current) return update;
   178→          return prev;
   179→        }
   180→        if (update.orderId !== prev.orderId) return prev;
   181→        return update;
   182→      });
   183→    });
   184→
   185→    sock.on('delivery_assigned', (assigned: DeliveryState) => {
   186→      if (!assigned) return;
   187→      setActiveOrderId(assigned.orderId);
   188→      setDelivery(assigned);
   189→      toast({
   190→        title: `${assigned.rider.name} assigned! 🏍️`,
   191→        description: `${assigned.rider.vehicle} • Rating ${assigned.rider.rating}★`,
   192→      });
   193→    });
   194→
   195→    sock.on('chat_history', (history: ChatMessage[]) => {
   196→      if (Array.isArray(history)) {
   197→        setMessages(history);
   198→      }
   199→    });
   200→
   201→    sock.on('new_message', (msg: ChatMessage) => {
   202→      if (!msg) return;
   203→      setMessages((prev) => {
   204→        // de-dupe by timestamp+text+from
   205→        if (
   206→          prev.some(
   207→            (m) =>
   208→              m.timestamp === msg.timestamp &&
   209→              m.text === msg.text &&
   210→              m.from === msg.from
   211→          )
   212→        ) {
   213→          return prev;
   214→        }
   215→        return [...prev, msg];
   216→      });
   217→    });
   218→
   219→    return () => {
   220→      sock.disconnect();
   221→      socketRef.current = null;
   222→      setIsConnected(false);
   223→    };
   224→  }, [isOpen]);
   225→
   226→  /* Auto-subscribe when activeOrderId changes (after first connect) */
   227→  useEffect(() => {
   228→    const sock = socketRef.current;
   229→    if (!sock || !isConnected || !activeOrderId) return;
   230→    sock.emit('subscribe_order', activeOrderId);
   231→  }, [isConnected, activeOrderId]);
   232→
   233→  /* Auto-scroll chat to bottom on new messages */
   234→  useEffect(() => {
   235→    if (chatScrollRef.current) {
   236→      chatScrollRef.current.scrollTop = chatScrollRef.current.scrollHeight;
   237→    }
   238→  }, [messages]);
   239→
   240→  /* ─── handlers ─── */
   241→  const handleClose = useCallback(() => {
   242→    setActiveModal(null);
   243→  }, [setActiveModal]);
   244→
   245→  const handleCallRider = useCallback(() => {
   246→    if (!delivery?.rider) return;
   247→    toast({
   248→      title: `Calling ${delivery.rider.name}... 📞`,
   249→      description: delivery.rider.phone,
   250→    });
   251→  }, [delivery, toast]);
   252→
   253→  const handleSendMessage = useCallback(() => {
   254→    const text = chatInput.trim();
   255→    const sock = socketRef.current;
   256→    if (!text || !sock || !delivery) return;
   257→    sock.emit('send_message', {
   258→      orderId: delivery.orderId,
   259→      from: 'customer',
   260→      text,
   261→    });
   262→    // Optimistically show our own message immediately
   263→    setMessages((prev) => {
   264→      if (
   265→        prev.some(
   266→          (m) =>
   267→            m.text === text &&
   268→            m.from === 'customer' &&
   269→            Date.now() - m.timestamp < 5000
   270→        )
   271→      ) {
   272→        return prev;
   273→      }
   274→      return [
   275→        ...prev,
   276→        {
   277→          orderId: delivery.orderId,
   278→          from: 'customer',
   279→          text,
   280→          timestamp: Date.now(),
   281→        },
   282→      ];
   283→    });
   284→    setChatInput('');
   285→  }, [chatInput, delivery]);
   286→
   287→  const handleRequestRider = useCallback(() => {
   288→    const sock = socketRef.current;
   289→    if (!sock) return;
   290→    setDelivery(null);
   291→    setMessages([]);
   292→    sock.emit('request_rider', {});
   293→    toast({
   294→      title: 'Finding a rider... 🌙',
   295→      description: 'Connecting you with the nearest SwiftRamadan rider',
   296→    });
   297→  }, [toast]);
   298→
   299→  /* ─── derived ─── */
   300→  const currentStageIndex = useMemo(() => {
   301→    if (!delivery) return 0;
   302→    const idx = STATUS_ORDER.indexOf(delivery.status);
   303→    return idx < 0 ? 0 : idx;
   304→  }, [delivery]);
   305→
   306→  // Map coordinates → pixel positions on the map panel (percentage)
   307→  const mapLayout = useMemo(() => {
   308→    if (!delivery) return null;
   309→    const pts = [delivery.store, delivery.customer, delivery.location];
   310→    const lats = pts.map((p) => p.lat);
   311→    const lngs = pts.map((p) => p.lng);
   312→    const minLat = Math.min(...lats);
   313→    const maxLat = Math.max(...lats);
   314→    const minLng = Math.min(...lngs);
   315→    const maxLng = Math.max(...lngs);
   316→    const latRange = Math.max(maxLat - minLat, 0.001);
   317→    const lngRange = Math.max(maxLng - minLng, 0.001);
   318→    const pad = 0.15; // 15% padding
   319→    const project = (p: GeoPoint) => {
   320→      const x = ((p.lng - minLng) / lngRange) * (1 - pad * 2) + pad;
   321→      const y = 1 - (((p.lat - minLat) / latRange) * (1 - pad * 2) + pad);
   322→      return { x: x * 100, y: y * 100 };
   323→    };
   324→    const store = project(delivery.store);
   325→    const customer = project(delivery.customer);
   326→    const rider = project(delivery.location);
   327→    return { store, customer, rider };
   328→  }, [delivery]);
   329→
   330→  return (
   331→    <AnimatePresence>
   332→      {isOpen && (
   333→        <>
   334→          {/* Backdrop */}
   335→          <motion.div
   336→            initial={{ opacity: 0 }}
   337→            animate={{ opacity: 1 }}
   338→            exit={{ opacity: 0 }}
   339→            className="fixed inset-0 bg-black/80 z-[95]"
   340→            onClick={handleClose}
   341→          />
   342→
   343→          {/* Full-screen modal */}
   344→          <motion.div
   345→            initial={{ y: '100%' }}
   346→            animate={{ y: 0 }}
   347→            exit={{ y: '100%' }}
   348→            transition={{ type: 'spring', damping: 28, stiffness: 220 }}
   349→            className="fixed inset-0 z-[100] bg-[#05070A] flex flex-col overflow-hidden"
   350→          >
   351→            {/* ─── Sticky Header ─── */}
   352→            <div className="glass-effect border-b border-white/5 px-4 py-3 flex items-center justify-between shrink-0 z-20">
   353→              <div className="flex items-center gap-3 min-w-0">
   354→                <div className="w-10 h-10 rounded-xl bg-[#13ec13]/10 border border-[#13ec13]/30 flex items-center justify-center shrink-0">
   355→                  <MapPin className="w-5 h-5 text-[#13ec13]" />
   356→                </div>
   357→                <div className="min-w-0">
   358→                  <h2 className="text-white text-base font-bold leading-tight flex items-center gap-2">
   359→                    Live Tracking
   360→                    {isConnected ? (
   361→                      <motion.span
   362→                        animate={{ opacity: [1, 0.4, 1] }}
   363→                        transition={{ duration: 1.5, repeat: Infinity }}
   364→                        className="w-2 h-2 rounded-full bg-[#13ec13] inline-block"
   365→                      />
   366→                    ) : (
   367→                      <span className="w-2 h-2 rounded-full bg-white/30 inline-block" />
   368→                    )}
   369→                  </h2>
   370→                  <p className="text-white/40 text-[11px] truncate">
   371→                    {delivery
   372→                      ? `Order #${delivery.orderId} • ${delivery.rider.name}`
   373→                      : 'Connecting to rider...'}
   374→                  </p>
   375→                </div>
   376→              </div>
   377→              <button
   378→                onClick={handleClose}
   379→                className="w-10 h-10 flex items-center justify-center rounded-full bg-white/5 hover:bg-white/10 transition-colors shrink-0"
   380→                aria-label="Close live tracking"
   381→              >
   382→                <X className="w-5 h-5 text-white/60" />
   383→              </button>
   384→            </div>
   385→
   386→            {/* ─── Scrollable content ─── */}
   387→            <div className="flex-1 overflow-y-auto custom-scrollbar">
   388→              {/* Map panel */}
   389→              <MapPanel
   390→                delivery={delivery}
   391→                layout={mapLayout}
   392→              />
   393→
   394→              {/* Status Timeline */}
   395→              <section className="px-4 pt-5">
   396→                <h3 className="text-white/80 text-xs font-bold uppercase tracking-wider mb-3">
   397→                  Delivery Status
   398→                </h3>
   399→                <div className="bg-[#0F1117] rounded-2xl border border-white/5 p-4">
   400→                  <div className="relative">
   401→                    {STATUS_STAGES.map((stage, idx) => {
   402→                      const isPast = idx < currentStageIndex;
   403→                      const isActive = idx === currentStageIndex;
   404→                      const isFuture = idx > currentStageIndex;
   405→                      const Icon = stage.icon;
   406→                      return (
   407→                        <div key={stage.key} className="flex gap-3 relative">
   408→                          {/* Vertical line */}
   409→                          {idx < STATUS_STAGES.length - 1 && (
   410→                            <div className="absolute left-[18px] top-9 bottom-0 w-[2px] bg-white/5" />
   411→                          )}
   412→                          {/* Marker */}
   413→                          <div
   414→                            className={`relative z-10 w-9 h-9 rounded-full flex items-center justify-center shrink-0 transition-all duration-500 ${
   415→                              isActive
   416→                                ? 'bg-[#13ec13]/20 border-2 border-[#13ec13] scale-110'
   417→                                : isPast
   418→                                  ? 'bg-[#13ec13] border-2 border-[#13ec13]'
   419→                                  : 'bg-[#1A1D26] border-2 border-white/10'
   420→                            }`}
   421→                          >
   422→                            {isPast ? (
   423→                              <Check className="w-4 h-4 text-[#05070A]" strokeWidth={3} />
   424→                            ) : (
   425→                              <Icon
   426→                                className={`w-4 h-4 ${
   427→                                  isActive ? 'text-[#13ec13]' : 'text-white/30'
   428→                                }`}
   429→                              />
   430→                            )}
   431→                            {isActive && (
   432→                              <motion.div
   433→                                animate={{ scale: [1, 1.6], opacity: [0.5, 0] }}
   434→                                transition={{ duration: 1.6, repeat: Infinity }}
   435→                                className="absolute inset-0 rounded-full bg-[#13ec13]"
   436→                              />
   437→                            )}
   438→                          </div>
   439→                          {/* Label */}
   440→                          <div className="flex-1 pb-5 pt-1">
   441→                            <div className="flex items-center gap-2">
   442→                              <p
   443→                                className={`text-sm font-bold ${
   444→                                  isActive
   445→                                    ? 'text-white'
   446→                                    : isPast
   447→                                      ? 'text-[#13ec13]'
   448→                                      : 'text-white/40'
   449→                                }`}
   450→                              >
   451→                                {stage.label}
   452→                              </p>
   453→                              {isActive && (
   454→                                <motion.span
   455→                                  animate={{ opacity: [1, 0.5, 1] }}
   456→                                  transition={{ duration: 1.4, repeat: Infinity }}
   457→                                  className="text-[10px] text-[#13ec13] font-bold uppercase"
   458→                                >
   459→                                  Live
   460→                                </motion.span>
   461→                              )}
   462→                            </div>
   463→                            <p
   464→                              className={`text-[11px] mt-0.5 ${
   465→                                isFuture ? 'text-white/30' : 'text-white/50'
   466→                              }`}
   467→                            >
   468→                              {stage.desc}
   469→                            </p>
   470→                          </div>
   471→                        </div>
   472→                      );
   473→                    })}
   474→                  </div>
   475→                </div>
   476→              </section>
   477→
   478→              {/* ETA + Rider cards */}
   479→              <section className="px-4 pt-4 grid grid-cols-1 sm:grid-cols-2 gap-3">
   480→                {/* ETA card */}
   481→                <div className="bg-gradient-to-br from-[#13ec13]/10 to-[#0F1117] rounded-2xl border border-[#13ec13]/20 p-4 relative overflow-hidden">
   482→                  <div className="absolute -right-4 -top-4 w-24 h-24 bg-[#13ec13]/5 blur-2xl rounded-full" />
   483→                  <div className="flex items-center gap-2 mb-2 relative">
   484→                    <Clock className="w-4 h-4 text-[#13ec13]" />
   485→                    <span className="text-white/50 text-[11px] uppercase tracking-wider font-bold">
   486→                      Estimated arrival
   487→                    </span>
   488→                  </div>
   489→                  <div className="flex items-baseline gap-2 relative">
   490→                    <motion.span
   491→                      key={delivery?.eta ?? 0}
   492→                      initial={{ scale: 1.2, opacity: 0 }}
   493→                      animate={{ scale: 1, opacity: 1 }}
   494→                      transition={{ duration: 0.3 }}
   495→                      className="text-white text-4xl font-black leading-none"
   496→                    >
   497→                      {delivery?.status === 'delivered' ? '0' : delivery?.eta ?? '--'}
   498→                    </motion.span>
   499→                    <span className="text-white/50 text-sm font-medium">
   500→                      {delivery?.status === 'delivered' ? 'Delivered' : 'min'}
   501→                    </span>
   502→                  </div>
   503→                  {delivery && (
   504→                    <p className="text-white/40 text-[11px] mt-2 relative">
   505→                      {delivery.status === 'delivered'
   506→                        ? 'Order completed 🎉'
   507→                        : delivery.status === 'arriving'
   508→                          ? 'Rider is at your doorstep!'
   509→                          : delivery.status === 'on_the_way'
   510→                            ? 'Rider approaching your area'
   511→                            : delivery.status === 'picked_up'
   512→                              ? 'Order picked up from restaurant'
   513→                              : 'Restaurant preparing your order'}
   514→                    </p>
   515→                  )}
   516→                </div>
   517→
   518→                {/* Rider card */}
   519→                <div className="bg-[#0F1117] rounded-2xl border border-white/5 p-4">
   520→                  {delivery ? (
   521→                    <div className="flex items-center gap-3">
   522→                      <div
   523→                        className="w-12 h-12 rounded-full flex items-center justify-center shrink-0 border-2 font-bold text-sm"
   524→                        style={{
   525→                          backgroundColor: `${delivery.rider.color}22`,
   526→                          borderColor: `${delivery.rider.color}66`,
   527→                          color: delivery.rider.color,
   528→                        }}
   529→                      >
   530→                        {initials(delivery.rider.name)}
   531→                      </div>
   532→                      <div className="flex-1 min-w-0">
   533→                        <p className="text-white text-sm font-bold truncate">
   534→                          {delivery.rider.name}
   535→                        </p>
   536→                        <div className="flex items-center gap-1">
   537→                          <Star className="w-3 h-3 fill-[#FFD700] text-[#FFD700]" />
   538→                          <span className="text-[#FFD700] text-[11px] font-bold">
   539→                            {delivery.rider.rating}
   540→                          </span>
   541→                          <span className="text-white/30 text-[11px]">•</span>
   542→                          <span className="text-white/40 text-[11px] truncate">
   543→                            {delivery.rider.vehicle}
   544→                          </span>
   545→                        </div>
   546→                      </div>
   547→                      <div className="flex gap-2 shrink-0">
   548→                        <button
   549→                          onClick={handleCallRider}
   550→                          className="w-10 h-10 rounded-xl bg-[#FFD700]/10 border border-[#FFD700]/30 flex items-center justify-center hover:bg-[#FFD700]/20 transition-colors"
   551→                          aria-label="Call rider"
   552→                        >
   553→                          <Phone className="w-4 h-4 text-[#FFD700]" />
   554→                        </button>
   555→                      </div>
   556→                    </div>
   557→                  ) : (
   558→                    <div className="flex items-center gap-3">
   559→                      <div className="w-12 h-12 rounded-full bg-[#1A1D26] border-2 border-white/10 flex items-center justify-center">
   560→                        <User className="w-5 h-5 text-white/30" />
   561→                      </div>
   562→                      <div className="flex-1">
   563→                        <p className="text-white/40 text-sm">Waiting for rider...</p>
   564→                        <button
   565→                          onClick={handleRequestRider}
   566→                          className="text-[#13ec13] text-xs font-bold mt-1 hover:underline"
   567→                        >
   568→                          Request a rider →
   569→                        </button>
   570→                      </div>
   571→                    </div>
   572→                  )}
   573→                </div>
   574→              </section>
   575→
   576→              {/* Live chat */}

   896→function ChatBubble({ msg, rider }: { msg: ChatMessage; rider?: Rider }) {
   897→  if (msg.from === 'system') {
   898→    return (
   899→      <div className="flex justify-center">
   900→        <span className="text-[10px] text-white/40 bg-white/5 px-3 py-1 rounded-full">
   901→          {msg.text}
   902→        </span>
   903→      </div>
   904→    );
   905→  }
   906→
   907→  const isCustomer = msg.from === 'customer';
   908→  const accentColor = rider?.color ?? '#13ec13';
   909→
   910→  return (
   911→    <motion.div
   912→      initial={{ opacity: 0, y: 8 }}
   913→      animate={{ opacity: 1, y: 0 }}
   914→      transition={{ duration: 0.2 }}
   915→      className={`flex items-end gap-2 ${isCustomer ? 'flex-row-reverse' : 'flex-row'}`}
   916→    >
   917→      {/* Avatar */}
   918→      <div
   919→        className="w-7 h-7 rounded-full flex items-center justify-center shrink-0 text-[10px] font-bold border"
   920→        style={{
   921→          backgroundColor: isCustomer ? '#13ec1322' : `${accentColor}22`,
   922→          borderColor: isCustomer ? '#13ec1366' : `${accentColor}66`,
   923→          color: isCustomer ? '#13ec13' : accentColor,
   924→        }}
   925→      >
   926→        {isCustomer ? (
   927→          'ME'
   928→        ) : (
   929→          <Bike className="w-3.5 h-3.5" />
   930→        )}
   931→      </div>
   932→      {/* Bubble */}
   933→      <div className={`max-w-[75%] ${isCustomer ? 'items-end' : 'items-start'} flex flex-col`}>
   934→        <div
   935→          className={`px-3 py-2 rounded-2xl text-sm ${
   936→            isCustomer
   937→              ? 'bg-[#13ec13] text-[#05070A] rounded-br-sm'
   938→              : 'bg-[#1A1D26] text-white rounded-bl-sm border border-white/5'
   939→          }`}
   940→        >
   941→          {msg.text}
   942→        </div>
   943→        <span className="text-[9px] text-white/30 mt-0.5 px-1">
   944→          {formatTime(msg.timestamp)}
   945→        </span>
   946→      </div>
   947→    </motion.div>
   948→  );
   949→}
