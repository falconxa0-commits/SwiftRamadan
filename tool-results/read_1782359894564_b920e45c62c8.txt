    18→const progressSteps = [
    19→  { key: 'confirmed', label: 'Confirmed', threshold: 10 },
    20→  { key: 'preparing', label: 'Preparing', threshold: 35 },
    21→  { key: 'ready', label: 'Ready', threshold: 55 },
    22→  { key: 'transit', label: 'In Transit', threshold: 75 },
    23→  { key: 'delivered', label: 'Delivered', threshold: 100 },
    24→];
    25→
    26→function OrderProgressTracker({ progress, status }: { progress: number; status: string }) {
    27→  return (
    28→    <div className="mt-3">
    29→      <div className="flex items-center justify-between">
    30→        {progressSteps.map((step, i) => {
    31→          const isActive = progress >= step.threshold;
    32→          const isCurrentStep = status === 'Preparing' && step.key === 'preparing' ||
    33→            status === 'In Transit' && step.key === 'transit' ||
    34→            status === 'Delivered' && step.key === 'delivered' ||
    35→            status === 'Confirmed' && step.key === 'confirmed' ||
    36→            status === 'Ready' && step.key === 'ready';
    37→          return (
    38→            <div key={step.key} className="flex items-center flex-1">
    39→              <div className="flex flex-col items-center">
    40→                <div className={`w-5 h-5 rounded-full flex items-center justify-center border-2 transition-colors ${
    41→                  isActive
    42→                    ? isCurrentStep
    43→                      ? 'border-[#10E07A] bg-[#10E07A]'
    44→                      : 'border-[#10E07A]/50 bg-[#10E07A]/20'
    45→                    : 'border-white/10 bg-[#0F1118]'
    46→                }`}>
    47→                  {isActive && (
    48→                    <div className={`w-1.5 h-1.5 rounded-full ${isCurrentStep ? 'bg-[#06070B]' : 'bg-[#10E07A]'}`} />
    49→                  )}
    50→                </div>
    51→                <span className={`text-[8px] mt-1 font-bold whitespace-nowrap ${
    52→                  isActive ? 'text-[#10E07A]' : 'text-white/20'
    53→                }`}>{step.label}</span>
    54→              </div>
    55→              {i < progressSteps.length - 1 && (
    56→                <div className={`flex-1 h-px mx-1 ${progress >= progressSteps[i + 1].threshold ? 'bg-[#10E07A]/40' : 'bg-white/5'}`} />
    57→              )}
    58→            </div>
    59→          );
    60→        })}
    61→      </div>
    62→    </div>
    63→  );
    64→}
