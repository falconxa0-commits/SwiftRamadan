    86→function formatRelativeTime(dateString: string): string {
    87→  try {
    88→    const date = new Date(dateString);
    89→    if (isNaN(date.getTime())) return 'just now';
    90→    const diffMs = Date.now() - date.getTime();
    91→    if (diffMs < 0) return 'just now';
    92→    const sec = Math.floor(diffMs / 1000);
    93→    if (sec < 60) return 'just now';
    94→    const min = Math.floor(sec / 60);
    95→    if (min < 60) return `${min}m ago`;
    96→    const hr = Math.floor(min / 60);
    97→    if (hr < 24) return `${hr}h ago`;
    98→    const day = Math.floor(hr / 24);
    99→    if (day < 7) return `${day}d ago`;
   100→    const week = Math.floor(day / 7);
   101→    if (week < 5) return `${week}w ago`;
   102→    const month = Math.floor(day / 30);
   103→    if (month < 12) return `${month}mo ago`;
   104→    const year = Math.floor(day / 365);
   105→    return `${year}y ago`;
   106→  } catch {
   107→    return 'just now';
   108→  }
   109→}

   111→function isLikedByMe(post: ApiPost, email: string): boolean {
   112→  const lb = post.likedBy;
   113→  if (!lb) return false;
   114→  if (Array.isArray(lb)) return lb.includes(email);
   115→  try {
   116→    const arr = JSON.parse(lb);
   117→    return Array.isArray(arr) && arr.includes(email);
   118→  } catch {
   119→    return false;
   120→  }
   121→}
   122→
   123→function likedByArray(post: ApiPost): string[] {
   124→  const lb = post.likedBy;
   125→  if (!lb) return [];
   126→  if (Array.isArray(lb)) return lb;
   127→  try {
   128→    const arr = JSON.parse(lb);
   129→    return Array.isArray(arr) ? arr.filter((x) => typeof x === 'string') : [];
   130→  } catch {
   131→    return [];
   132→  }
   133→}

   135→// Convert mock communityPosts (from @/lib/data) into ApiPost shape for fallback.
   136→function seedToApiPosts(): ApiPost[] {
   137→  const now = Date.now();
   138→  return communityPosts.map((p, idx) => {
   139→    const created = new Date(now - (idx + 1) * 60 * 60 * 1000).toISOString();
   140→    return {
   141→      id: `seed-${p.id}`,
   142→      authorName: p.author,
   143→      authorInitial: p.avatar,
   144→      category: p.category,
   145→      content: p.content,
   146→      imageUrl: null,
   147→      likes: p.likes,
   148→      likedBy: [],
   149→      createdAt: created,
   150→      comments: Array.from({ length: Math.min(p.replies, 2) }).map((_, ci) => ({
   151→        id: `seed-c-${p.id}-${ci}`,
   152→        authorName: ci === 0 ? 'Sister Khadija' : 'Brother Yusuf',
   153→        authorInitial: ci === 0 ? 'S' : 'Y',
   154→        content:
   155→          ci === 0
   156→            ? 'Subhanallah, thank you for sharing! 🤲'
   157→            : 'Jazak Allahu khairan for the tip 🌙',
   158→        createdAt: new Date(
   159→          now - (idx + 1) * 60 * 60 * 1000 + (ci + 1) * 5 * 60 * 1000,
   160→        ).toISOString(),
   161→      })),
   162→    };
   163→  });
   164→}

   216→        setPosts(data.posts);
   217→      } else {
   218→        setPosts(seedToApiPosts());
   219→      }
   220→    } catch {
   221→      if (reqId !== reqIdRef.current) return;
   222→      setLoadError('Could not reach the community feed. Showing sample posts.');
   223→      setPosts(seedToApiPosts());

   225→      if (reqId === reqIdRef.current) setLoading(false);
   226→    }
   227→  }, []);
   228→
   229→  useEffect(() => {
   230→    if (!isOpen) return;

   262→        prev.map((p) =>
   263→          p.id === post.id ? { ...p, likes: newCount, likedBy: newArr } : p,

   348→            email,

   375→          prev.map((p) =>
   376→            p.id === post.id

   378→                  ...p,

   388→          const next = { ...prev };
   389→          delete next[post.id];
