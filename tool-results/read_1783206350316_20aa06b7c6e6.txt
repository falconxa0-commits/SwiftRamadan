   160→const GENERIC_STEPS = [
   161→  'Prep ingredients: wash, chop, and measure everything before you start.',
   162→  'Heat oil in a pan over medium heat until shimmering.',
   163→  'Cook the base: sauté onions and aromatics until fragrant.',

   168→
   169→function generateSteps(recipeName: string): string[] {
   170→  const n = recipeName.toLowerCase();
   171→  const steps = [...GENERIC_STEPS];
   172→  if (n.includes('smoothie') || n.includes('drink') || n.includes('juice')) {
   173→    return [
   174→      'Prep ingredients: wash fruits and measure liquids.',
   175→      'Add base liquids (milk/water/yogurt) to the blender first.',
   176→      'Toss in fruits, dates, and nuts.',
   177→      'Blend on high for 45–60 seconds until silky.',
   178→      'Taste, sweeten if needed, and pour over ice.',
   179→    ];
   180→  }
   181→  if (n.includes('suya') || n.includes('grill')) {
   182→    return [
   183→      'Prep & cut protein into thin strips; thread onto skewers.',
   184→      'Mix suya spice (kuli-kuli, paprika, ginger, salt) on a plate.',
   185→      'Coat skewers generously with the spice mix.',
   186→      'Grill over high heat, 2–3 min per side, until charred.',
   187→      'Rest 2 minutes, then serve with onions and tomatoes.',
   188→    ];
   189→  }
   190→  // default 6-step generic flow
   191→  return steps;
   192→}
   193→
   194→function difficultyColor(d: string): string {
   195→  if (d === 'easy') return '#13ec13';
   196→  if (d === 'hard') return '#ef4444';
   197→  return '#FFD700';
   198→}

   360→            image,
   361→            recipeName: recipe.name,

   363→            stepIndex: idx,
   364→            email: mail,
   365→          }),

   370→        }
   371→      } catch {
   372→        /* silent — next tick will retry */
   373→      } finally {
   374→        setVisionLoading(false);
   375→      }
   376→    };
   377→
   378→    // fire one immediately so the user gets feedback fast, then every 5s
   379→    tick();

   381→    intervalRef.current = id;
   382→
   383→    return () => {
   384→      clearInterval(id);
