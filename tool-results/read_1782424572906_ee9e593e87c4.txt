    34→function ConfettiParticle({ delay, color }: { delay: number; color: string }) {
    35→  const randomX = Math.random() * 300 - 150;
    36→  const randomY = Math.random() * 400 + 100;
    37→  const randomRotate = Math.random() * 720 - 360;
    38→  const size = Math.random() * 8 + 4;
    39→
    40→  return (
    41→    <motion.div
    42→      initial={{ opacity: 1, x: 0, y: 0, rotate: 0, scale: 1 }}
    43→      animate={{ opacity: 0, x: randomX, y: randomY, rotate: randomRotate, scale: 0.3 }}
    44→      transition={{ duration: 1.8, delay, ease: 'easeOut' }}
    45→      className="absolute rounded-sm pointer-events-none"
    46→      style={{
    47→        width: size,
    48→        height: size,
    49→        backgroundColor: color,
    50→        left: '50%',
    51→        top: '30%',
    52→      }}
    53→    />
    54→  );
    55→}
