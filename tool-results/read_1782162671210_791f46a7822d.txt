    41→/**
    42→ * Reads an image File and returns a compressed JPEG data URL.
    43→ * Draws to an offscreen canvas at MAX_WIDTH (preserving aspect ratio),
    44→ * exports as JPEG quality 0.8.
    45→ */
    46→function compressImage(file: File): Promise<string> {
    47→  return new Promise((resolve, reject) => {
    48→    const reader = new FileReader();
    49→    reader.onerror = () => reject(new Error('FileReader failed'));
    50→    reader.onload = () => {
    51→      const dataUrl = reader.result as string;
    52→      const img = new Image();
    53→      img.onerror = () => reject(new Error('Image load failed'));
    54→      img.onload = () => {
    55→        let { width, height } = img;
    56→        if (width > MAX_WIDTH) {
    57→          height = Math.round((height * MAX_WIDTH) / width);
    58→          width = MAX_WIDTH;
    59→        }
    60→        const canvas = document.createElement('canvas');
    61→        canvas.width = width;
    62→        canvas.height = height;
    63→        const ctx = canvas.getContext('2d');
    64→        if (!ctx) {
    65→          // Fallback to original data URL if canvas is unavailable
    66→          resolve(dataUrl);
    67→          return;
    68→        }
    69→        ctx.drawImage(img, 0, 0, width, height);
    70→        try {
    71→          const compressed = canvas.toDataURL('image/jpeg', 0.8);
    72→          resolve(compressed);
    73→        } catch {
    74→          resolve(dataUrl);
    75→        }
    76→      };
    77→      img.src = dataUrl;
    78→    };
    79→    reader.readAsDataURL(file);
    80→  });
    81→}

    83→function findSimilarProducts(
    84→  result: VisualSearchResult,
    85→  limit = 4
    86→) {
    87→  const foodNameLower = result.foodName.toLowerCase();
    88→  const tags = result.tags || [];
    89→
    90→  // Map the VLM category to the lowercase category strings stored in allProducts.
    91→  // allProducts categories are like 'iftar meals', 'sahur', 'dates', 'drinks',
    92→  // 'snacks', 'fruits', 'groceries', 'bundles', 'flash-sale', 'charity', etc.
    93→  const targetCategory = result.category.toLowerCase();
    94→
    95→  const scored = allProducts
    96→    .filter((p) => p.inStock !== false)
    97→    .map((p) => {
    98→      const pName = p.name.toLowerCase();
    99→      const pCategory = String(p.category || '').toLowerCase();
   100→      let score = 0;
   101→
   102→      // Category match (highest weight)
   103→      if (pCategory === targetCategory) score += 50;
   104→      else if (pCategory.includes(targetCategory) || targetCategory.includes(pCategory))
   105→        score += 25;
   106→
   107→      // Tag matches against product name
   108→      for (const tag of tags) {
   109→        if (tag && pName.includes(tag)) score += 15;
   110→      }
   111→
   112→      // Food name tokens against product name
   113→      const nameTokens = foodNameLower
   114→        .split(/\s+/)
   115→        .filter((t) => t.length > 2);
   116→      for (const token of nameTokens) {
   117→        if (pName.includes(token)) score += 10;
   118→      }
   119→
   120→      // Tag matches against category
   121→      for (const tag of tags) {
   122→        if (tag && pCategory.includes(tag)) score += 5;
   123→      }
   124→
   125→      return { product: p, score };
   126→    })
   127→    .filter((s) => s.score > 0)
   128→    .sort((a, b) => b.score - a.score)
   129→    .slice(0, limit)
   130→    .map((s) => s.product);
   131→
   132→  // If nothing matched, return the first few products so the section isn't empty
   133→  if (scored.length === 0) {
   134→    return allProducts.slice(0, limit);
   135→  }
   136→  return scored;
   137→}

   546→                                <span className="absolute top-1.5 left-1.5 px-1.5 py-0.5 rounded-md bg-red-500 text-white text-[9px] font-black">
   547→                                  -
   548→                                  {Math.round(
   549→                                    (1 -
   550→                                      product.salePrice /
   551→                                        product.originalPrice) *
   552→                                      100
   553→                                  )}
   554→                                  %
   555→                                </span>
   556→                              )}
   557→                            </div>
   558→                            <div className="p-2.5 flex flex-col gap-1.5 flex-1">
   559→                              <h5 className="text-white text-xs font-bold leading-tight line-clamp-2">
   560→                                {product.name}
   561→                              </h5>
   562→                              <div className="flex items-center gap-1 text-[10px] text-white/40">
   563→                                <span>★ {product.rating?.toFixed(1) ?? '4.5'}</span>
   564→                                <span>•</span>
   565→                                <span>{product.deliveryTime ?? '20 min'}</span>
   566→                              </div>
   567→                              <div className="flex items-center justify-between gap-2 mt-auto pt-1">
   568→                                <span className="text-[#13ec13] font-black text-sm">
   569→                                  {formatNaira(price)}
   570→                                </span>
   571→                                <button
   572→                                  onClick={() => handleAddToCart(product)}
   573→                                  aria-label={`Add ${product.name} to cart`}
   574→                                  className={`shrink-0 w-7 h-7 rounded-lg flex items-center justify-center transition-all ${
   575→                                    inCart
   576→                                      ? 'bg-[#13ec13]/20 text-[#13ec13] border border-[#13ec13]/40'
   577→                                      : 'bg-[#13ec13] text-[#05070A] active:scale-95 hover:shadow-md hover:shadow-[#13ec13]/30'
   578→                                  }`}
   579→                                >
   580→                                  {inCart ? (
   581→                                    <Check className="w-4 h-4" />
   582→                                  ) : (
   583→                                    <Plus />
   584→                                  )}
   585→                                </button>
   586→                              </div>
   587→                            </div>
   588→                          </motion.div>
   589→                        );
   590→                      })}
   591→                    </div>
   592→                  </div>
   593→
   594→                  {/* Try another photo */}
   595→                  <button
   596→                    onClick={reset}
   597→                    className="w-full py-3.5 rounded-2xl bg-white/5 text-white font-bold text-sm flex items-center justify-center gap-2 border border-white/10 hover:border-[#13ec13]/30 hover:bg-white/10 transition-colors"
   598→                  >
   599→                    <RefreshCw className="w-4 h-4 text-[#13ec13]" />
   600→                    Try Another Photo
   601→                  </button>
   602→                </motion.div>
   603→              )}
   604→            </div>
   605→
   606→            {/* Hidden inputs */}
   607→            <input
   608→              ref={cameraInputRef}
   609→              type="file"
   610→              accept="image/*"
   611→              capture="environment"
   612→              className="hidden"
   613→              onChange={(e) => {
   614→                const file = e.target.files?.[0];
   615→                handleFileSelected(file);
   616→              }}
   617→            />
   618→            <input
   619→              ref={fileInputRef}
   620→              type="file"
   621→              accept="image/*"
   622→              className="hidden"
   623→              onChange={(e) => {
   624→                const file = e.target.files?.[0];
   625→                handleFileSelected(file);
   626→              }}
   627→            />
   628→          </motion.div>
   629→        </>
   630→      )}
   631→    </AnimatePresence>
   632→  );
   633→}
