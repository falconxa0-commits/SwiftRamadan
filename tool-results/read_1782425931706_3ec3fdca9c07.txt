     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  Plus,
     7→  Pencil,
     8→  AlertTriangle,
     9→  ShoppingBag,
    10→  Package,
    11→  TrendingUp,
    12→  Star,
    13→  Eye,
    14→  Trash2,
    15→  Loader2,
    16→  X,
    17→  Check,
    18→} from 'lucide-react';
    19→import { formatNaira } from '@/lib/data';
    20→import { useAppStore } from '@/lib/store';
    21→import { useToast } from '@/hooks/use-toast';
    22→
    23→/* ──────────────────── Types ──────────────────── */
    24→
    25→type VendorProduct = {
    26→  id: string;
    27→  name: string;
    28→  description?: string;
    29→  price: number;
    30→  salePrice?: number;
    31→  originalPrice?: number;
    32→  category: string;
    33→  rating: number;
    34→  reviewCount: number;
    35→  deliveryTime: string;
    36→  inStock: boolean;
    37→  image: string;
    38→  images: string[];
    39→  vendorId?: string;
    40→  createdAt?: string;
    41→};
    42→
    43→type CategoryFilter = 'All' | 'meals' | 'snacks' | 'drinks' | 'desserts' | 'groceries';
    44→
    45→/* ════════════════════════════════════════════════════════════════
    46→   MAIN COMPONENT
    47→   ════════════════════════════════════════════════════════════════ */
    48→
    49→export default function VendorStoreTab() {
    50→  const { toast } = useToast();
    51→  const { setActiveModal, userEmail } = useAppStore();
    52→  const [activeCategory, setActiveCategory] = useState<CategoryFilter>('All');
    53→  const [products, setProducts] = useState<VendorProduct[]>([]);
    54→  const [loading, setLoading] = useState(true);
    55→  const [editingId, setEditingId] = useState<string | null>(null);
    56→  const [editForm, setEditForm] = useState<Partial<VendorProduct>>({});
    57→  const [confirmDeleteId, setConfirmDeleteId] = useState<string | null>(null);
    58→  const [busyId, setBusyId] = useState<string | undefined>(undefined);
    59→
    60→  const categories: CategoryFilter[] = ['All', 'meals', 'snacks', 'drinks', 'desserts', 'groceries'];
    61→
    62→  /* ── Fetch products ── */
    63→  const fetchProducts = useCallback(async () => {
    64→    setLoading(true);
    65→    try {
    66→      const res = await fetch(
    67→        `/api/vendor/products?vendorEmail=${encodeURIComponent(userEmail || 'sani@swiftramadan.app')}`
    68→      );
    69→      const json = await res.json();
    70→      if (json.success && Array.isArray(json.products)) {
    71→        setProducts(json.products);
    72→      } else {
    73→        setProducts([]);
    74→      }
    75→    } catch (err) {
    76→      console.error('[VendorStoreTab] fetch error:', err);
    77→      toast({
    78→        title: 'Failed to load products',
    79→        description: 'Please try again later.',
    80→        variant: 'destructive',
    81→      });
    82→    } finally {
    83→      setLoading(false);
    84→    }
    85→  }, [userEmail, toast]);
    86→
    87→  useEffect(() => {
    88→    fetchProducts();
    89→  }, [fetchProducts]);
    90→
    91→  // Listen for cross-component refresh signals (e.g. when VendorAddProductModal adds a product)
    92→  useEffect(() => {
    93→    const onChange = () => fetchProducts();
    94→    if (typeof window !== 'undefined') {
    95→      window.addEventListener('vendor-products-changed', onChange);
    96→    }
    97→    return () => {
    98→      if (typeof window !== 'undefined') {
    99→        window.removeEventListener('vendor-products-changed', onChange);
   100→      }
   101→    };
   102→  }, [fetchProducts]);
   103→
   104→  /* ── Filtered items ── */
   105→  const filteredItems =
   106→    activeCategory === 'All' ? products : products.filter((p) => p.category === activeCategory);
   107→
   108→  const unavailableCount = products.filter((p) => !p.inStock).length;
   109→  const totalOrders = products.reduce((sum, _p) => sum + 0, 0); // orders count not in DB schema; placeholder
   110→
   111→  /* ── Toggle inStock ── */
   112→  const toggleAvailability = async (product: VendorProduct) => {
   113→    setBusyId(product.id);
   114→    const nextStock = !product.inStock;
   115→    // Optimistic update
   116→    setProducts((prev) =>
   117→      prev.map((p) => (p.id === product.id ? { ...p, inStock: nextStock } : p))
   118→    );
   119→    try {
   120→      const res = await fetch(
   121→        `/api/vendor/products?id=${product.id}&vendorEmail=${encodeURIComponent(userEmail || '')}`,
   122→        {
   123→          method: 'PUT',
   124→          headers: { 'Content-Type': 'application/json' },
   125→          body: JSON.stringify({ inStock: nextStock }),
   126→        }
   127→      );
   128→      const json = await res.json();
   129→      if (!json.success) throw new Error(json.error || 'Update failed');
   130→      toast({
   131→        title: nextStock ? 'Item Available ✅' : 'Item Unavailable ❌',
   132→        description: `${product.name} is now ${nextStock ? 'visible to customers' : 'hidden from customers'}`,
   133→      });
   134→    } catch (err) {
   135→      console.error(err);
   136→      // Revert
   137→      setProducts((prev) =>
   138→        prev.map((p) => (p.id === product.id ? { ...p, inStock: !nextStock } : p))
   139→      );
   140→      toast({
   141→        title: 'Update failed',
   142→        description: 'Could not update availability.',
   143→        variant: 'destructive',
   144→      });
   145→    } finally {
   146→      setBusyId(undefined);
   147→    }
   148→  };
   149→
   150→  /* ── Start inline edit ── */
   151→  const startEdit = (product: VendorProduct) => {
   152→    setEditingId(product.id);
   153→    setEditForm({
   154→      name: product.name,
   155→      description: product.description,
   156→      price: product.price,
   157→      category: product.category,
   158→      deliveryTime: product.deliveryTime,
   159→      image: product.image,
   160→    });
   161→  };
   162→
   163→  const cancelEdit = () => {
   164→    setEditingId(null);
   165→    setEditForm({});
   166→  };
   167→
   168→  /* ── Save edit ── */
   169→  const saveEdit = async (productId: string) => {
   170→    setBusyId(productId);
   171→    try {
   172→      const res = await fetch(
   173→        `/api/vendor/products?id=${productId}&vendorEmail=${encodeURIComponent(userEmail || '')}`,
   174→        {
   175→          method: 'PUT',
   176→          headers: { 'Content-Type': 'application/json' },
   177→          body: JSON.stringify({
   178→            name: editForm.name,
   179→            description: editForm.description,
   180→            price: Number(editForm.price) || 0,
   181→            category: editForm.category,
   182→            deliveryTime: editForm.deliveryTime,
   183→            image: editForm.image,
   184→          }),
   185→        }
   186→      );
   187→      const json = await res.json();
   188→      if (!json.success) throw new Error(json.error || 'Update failed');
   189→      // Update local state
   190→      setProducts((prev) =>
   191→        prev.map((p) => (p.id === productId ? { ...p, ...json.product } : p))
   192→      );
   193→      toast({
   194→        title: 'Product Updated ✅',
   195→        description: `${json.product.name} has been updated`,
   196→      });
   197→      cancelEdit();
   198→    } catch (err) {
   199→      console.error(err);
   200→      toast({
   201→        title: 'Update failed',
   202→        description: 'Could not save changes.',
   203→        variant: 'destructive',
   204→      });
   205→    } finally {
   206→      setBusyId(undefined);
   207→    }
   208→  };
   209→
   210→  /* ── Delete product ── */
   211→  const confirmDelete = async () => {
   212→    if (!confirmDeleteId) return;
   213→    setBusyId(confirmDeleteId);
   214→    try {
   215→      const res = await fetch(
   216→        `/api/vendor/products?id=${confirmDeleteId}&vendorEmail=${encodeURIComponent(userEmail || '')}`,
   217→        { method: 'DELETE' }
   218→      );
   219→      const json = await res.json();
   220→      if (!json.success) throw new Error(json.error || 'Delete failed');
   221→      setProducts((prev) => prev.filter((p) => p.id !== confirmDeleteId));
   222→      toast({
   223→        title: 'Product Deleted',
   224→        description: 'Item removed from your menu',
   225→        variant: 'destructive',
   226→      });
   227→    } catch (err) {
   228→      console.error(err);
   229→      toast({
   230→        title: 'Delete failed',
   231→        description: 'Could not delete product.',
   232→        variant: 'destructive',
   233→      });
   234→    } finally {
   235→      setBusyId(undefined);
   236→      setConfirmDeleteId(null);
   237→    }
   238→  };
   239→
   240→  /* ── Render ── */
   241→  return (
   242→    <main className="flex-1 overflow-y-auto pb-32">
   243→      <div className="flex flex-col gap-5 px-4 pt-2">
   244→        {/* Stock Alert */}
   245→        <AnimatePresence>
   246→          {unavailableCount > 0 && (
   247→            <motion.div
   248→              initial={{ opacity: 0, y: -10, height: 0 }}
   249→              animate={{ opacity: 1, y: 0, height: 'auto' }}
   250→              exit={{ opacity: 0, y: -10, height: 0 }}
   251→              className="rounded-2xl bg-red-500/10 border border-red-500/20 p-4"
   252→            >
   253→              <div className="flex items-center gap-3">
   254→                <div className="w-9 h-9 rounded-xl bg-red-500/20 flex items-center justify-center border border-red-500/20 shrink-0">
   255→                  <AlertTriangle className="w-4 h-4 text-red-400" />
   256→                </div>
   257→                <div>
   258→                  <p className="text-red-400 text-sm font-bold">Stock Alert</p>
   259→                  <p className="text-white/40 text-xs mt-0.5">
   260→                    {unavailableCount} item{unavailableCount > 1 ? 's' : ''} currently unavailable —
   261→                    customers can&apos;t order them
   262→                  </p>
   263→                </div>
   264→              </div>
   265→            </motion.div>
   266→          )}
   267→        </AnimatePresence>
   268→
   269→        {/* Quick Stats */}
   270→        <motion.div
   271→          initial={{ opacity: 0, y: -5 }}
   272→          animate={{ opacity: 1, y: 0 }}
   273→          className="grid grid-cols-3 gap-3"
   274→        >
   275→          <div className="bg-[#0F1118] rounded-xl p-3 border border-white/5 text-center">
   276→            <p className="text-[#F5C451] text-lg font-black">{products.length}</p>
   277→            <p className="text-white/30 text-[9px] font-bold uppercase">Menu Items</p>
   278→          </div>
   279→          <div className="bg-[#0F1118] rounded-xl p-3 border border-white/5 text-center">
   280→            <p className="text-[#10E07A] text-lg font-black">{products.filter((i) => i.inStock).length}</p>
   281→            <p className="text-white/30 text-[9px] font-bold uppercase">Available</p>
   282→          </div>
   283→          <div className="bg-[#0F1118] rounded-xl p-3 border border-white/5 text-center">
   284→            <p className="text-white text-lg font-black">{totalOrders.toLocaleString()}</p>
   285→            <p className="text-white/30 text-[9px] font-bold uppercase">Total Orders</p>
   286→          </div>
   287→        </motion.div>
   288→
   289→        {/* Section Header */}
   290→        <motion.div
   291→          initial={{ opacity: 0, y: -5 }}
   292→          animate={{ opacity: 1, y: 0 }}
   293→          className="flex items-center justify-between"
   294→        >
   295→          <div className="flex items-center gap-2">
   296→            <h2 className="text-white text-lg font-black">Menu Items</h2>
   297→            <span className="px-2 py-0.5 rounded-full bg-[#F5C451]/20 text-[#F5C451] text-[10px] font-black border border-[#F5C451]/20">
   298→              {products.length}
   299→            </span>
   300→          </div>
   301→          <button
   302→            onClick={() => setActiveModal('vendor-stock')}
   303→            className="text-[#F5C451] text-xs font-bold flex items-center gap-1 hover:opacity-80 transition-opacity"
   304→          >
   305→            Stock Control <Package className="w-3.5 h-3.5" />
   306→          </button>
   307→        </motion.div>
   308→
   309→        {/* Add Product CTA (top) */}
   310→        <motion.button
   311→          initial={{ opacity: 0, y: 5 }}
   312→          animate={{ opacity: 1, y: 0 }}
   313→          onClick={() => useAppStore.getState().setActiveModal('vendor-add-product')}
   314→          className="w-full py-3 rounded-2xl bg-[#F5C451] text-[#06070B] text-sm font-bold hover:bg-[#F5C451]/90 active:scale-[0.98] transition-all flex items-center justify-center gap-2 gold-glow"
   315→        >
   316→          <Plus className="w-4 h-4" strokeWidth={3} />
   317→          Add New Product
   318→        </motion.button>
   319→
   320→        {/* Category Filter Chips */}
   321→        <motion.div
   322→          initial={{ opacity: 0, y: 5 }}
   323→          animate={{ opacity: 1, y: 0 }}
   324→          transition={{ delay: 0.05 }}
   325→          className="flex gap-2 overflow-x-auto no-scrollbar -mx-4 px-4"
   326→        >
   327→          {categories.map((cat) => (
   328→            <button
   329→              key={cat}
   330→              onClick={() => setActiveCategory(cat)}
   331→              className={`px-4 py-2 rounded-full text-xs font-bold whitespace-nowrap transition-all shrink-0 capitalize ${
   332→                activeCategory === cat
   333→                  ? 'bg-[#F5C451]/20 text-[#F5C451] border border-[#F5C451]/30'
   334→                  : 'bg-[#0F1118] text-white/40 border border-white/5 hover:bg-white/10'
   335→              }`}
   336→            >
   337→              {cat}
   338→            </button>
   339→          ))}
   340→        </motion.div>
   341→
   342→        {/* Loading state */}
   343→        {loading ? (
   344→          <div className="flex flex-col items-center py-16 text-center">
   345→            <Loader2 className="w-8 h-8 text-[#F5C451] animate-spin mb-3" />
   346→            <p className="text-white/40 text-sm font-semibold">Loading your menu...</p>
   347→          </div>
   348→        ) : filteredItems.length === 0 ? (
   349→          /* Empty state */
   350→          <div className="flex flex-col items-center py-16 text-center">
   351→            <div className="w-16 h-16 rounded-full bg-white/5 flex items-center justify-center mb-4">
   352→              <Package className="w-8 h-8 text-white/20" />
   353→            </div>
   354→            <p className="text-white/40 text-sm font-semibold">No items yet</p>
   355→            <p className="text-white/20 text-xs mt-1">
   356→              Add your first product to start selling this Ramadan
   357→            </p>
   358→            <button
   359→              onClick={() => useAppStore.getState().setActiveModal('vendor-add-product')}
   360→              className="mt-4 px-5 py-2.5 rounded-xl bg-[#F5C451]/20 text-[#F5C451] text-xs font-bold border border-[#F5C451]/30 hover:bg-[#F5C451]/30 transition-all flex items-center gap-1.5"
   361→            >
   362→              <Plus className="w-3.5 h-3.5" strokeWidth={3} />
   363→              Add Product
   364→            </button>
   365→          </div>
   366→        ) : (
   367→          /* Menu Item Cards */
   368→          <div className="space-y-3">
   369→            <AnimatePresence mode="popLayout">
   370→              {filteredItems.map((item, i) => {
   371→                const isEditing = editingId === item.id;
   372→                const isBusy = busyId === item.id;
   373→                const isConfirmDelete = confirmDeleteId === item.id;
   374→
   375→                return (
   376→                  <motion.div
   377→                    key={item.id}
   378→                    layout
   379→                    initial={{ opacity: 0, y: 15 }}
   380→                    animate={{ opacity: 1, y: 0 }}
   381→                    exit={{ opacity: 0, scale: 0.95 }}
   382→                    transition={{ delay: i * 0.05 }}
   383→                    className={`rounded-2xl bg-[#0F1118] border p-4 transition-all ${
   384→                      item.inStock ? 'border-white/5' : 'border-red-500/20 opacity-70'
   385→                    }`}
   386→                  >
   387→                    {isEditing ? (
   388→                      /* ── Inline Edit Form ── */
   389→                      <div className="space-y-3">
   390→                        <div className="flex items-center justify-between mb-1">
   391→                          <h3 className="text-[#F5C451] text-sm font-bold">Edit Item</h3>
   392→                          <button
   393→                            onClick={cancelEdit}
   394→                            className="text-white/30 text-xs hover:text-white/50 transition-colors"
   395→                          >
   396→                            Cancel
   397→                          </button>
   398→                        </div>
   399→                        <input
   400→                          type="text"
   401→                          placeholder="Item name"
   402→                          value={editForm.name || ''}
   403→                          onChange={(e) => setEditForm((f) => ({ ...f, name: e.target.value }))}
   404→                          className="w-full bg-[#06070B]/50 border border-white/10 rounded-xl px-4 py-2.5 text-white text-sm placeholder:text-white/20 focus:border-[#F5C451]/30 focus:outline-none transition-colors"
   405→                        />
   406→                        <textarea
   407→                          placeholder="Description"
   408→                          value={editForm.description || ''}
   409→                          onChange={(e) => setEditForm((f) => ({ ...f, description: e.target.value }))}
   410→                          rows={2}
   411→                          className="w-full bg-[#06070B]/50 border border-white/10 rounded-xl px-4 py-2.5 text-white text-sm placeholder:text-white/20 focus:border-[#F5C451]/30 focus:outline-none transition-colors resize-none"
   412→                        />
   413→                        <div className="grid grid-cols-2 gap-3">
   414→                          <input
   415→                            type="number"
   416→                            placeholder="Price (₦)"
   417→                            value={editForm.price ?? ''}
   418→                            onChange={(e) =>
   419→                              setEditForm((f) => ({ ...f, price: Number(e.target.value) }))
   420→                            }
   421→                            className="w-full bg-[#06070B]/50 border border-white/10 rounded-xl px-4 py-2.5 text-white text-sm placeholder:text-white/20 focus:border-[#F5C451]/30 focus:outline-none transition-colors"
   422→                          />
   423→                          <select
   424→                            value={editForm.category || 'meals'}
   425→                            onChange={(e) => setEditForm((f) => ({ ...f, category: e.target.value }))}
   426→                            className="w-full bg-[#06070B]/50 border border-white/10 rounded-xl px-4 py-2.5 text-white text-sm focus:border-[#F5C451]/30 focus:outline-none transition-colors"
   427→                          >
   428→                            {categories.filter((c) => c !== 'All').map((c) => (
   429→                              <option key={c} value={c} className="bg-[#0F1118]">
   430→                                {c}
   431→                              </option>
   432→                            ))}
   433→                          </select>
   434→                        </div>
   435→                        <input
   436→                          type="text"
   437→                          placeholder="Image URL"
   438→                          value={editForm.image || ''}
   439→                          onChange={(e) => setEditForm((f) => ({ ...f, image: e.target.value }))}
   440→                          className="w-full bg-[#06070B]/50 border border-white/10 rounded-xl px-4 py-2.5 text-white text-sm placeholder:text-white/20 focus:border-[#F5C451]/30 focus:outline-none transition-colors"
   441→                        />
   442→                        <input
   443→                          type="text"
   444→                          placeholder="Delivery time (e.g. 25 min)"
   445→                          value={editForm.deliveryTime || ''}
   446→                          onChange={(e) => setEditForm((f) => ({ ...f, deliveryTime: e.target.value }))}
   447→                          className="w-full bg-[#06070B]/50 border border-white/10 rounded-xl px-4 py-2.5 text-white text-sm placeholder:text-white/20 focus:border-[#F5C451]/30 focus:outline-none transition-colors"
   448→                        />
   449→                        <button
   450→                          onClick={() => saveEdit(item.id)}
   451→                          disabled={isBusy}
   452→                          className="w-full py-3 rounded-xl bg-[#F5C451] text-[#06070B] text-sm font-bold hover:bg-[#F5C451]/90 active:scale-[0.98] transition-all flex items-center justify-center gap-1.5 disabled:opacity-60"
   453→                        >
   454→                          {isBusy ? (
   455→                            <Loader2 className="w-4 h-4 animate-spin" />
   456→                          ) : (
   457→                            <Check className="w-4 h-4" strokeWidth={3} />
   458→                          )}
   459→                          Save Changes
   460→                        </button>
   461→                      </div>
   462→                    ) : (
   463→                      /* ── Normal Item View ── */
   464→                      <>
   465→                        <div className="flex items-start gap-3">
   466→                          {/* Item Image Thumbnail */}
   467→                          <div
   468→                            className="w-16 h-16 rounded-xl bg-cover bg-center shrink-0 border border-white/10 bg-white/5"
   469→                            style={
   470→                              item.image
   471→                                ? { backgroundImage: `url(${item.image})` }
   472→                                : undefined
   473→                            }
   474→                          >
   475→                            {!item.image && (
   476→                              <div className="w-full h-full flex items-center justify-center">
   477→                                <Package className="w-5 h-5 text-white/20" />
   478→                              </div>
   479→                            )}
   480→                          </div>
   481→
   482→                          {/* Item Details */}
   483→                          <div className="flex-1 min-w-0">
   484→                            <div className="flex items-start justify-between gap-2">
   485→                              <div className="min-w-0">
   486→                                <p
   487→                                  className={`text-sm font-bold truncate ${
   488→                                    item.inStock ? 'text-white' : 'text-white/50'
   489→                                  }`}
   490→                                >
   491→                                  {item.name}
   492→                                </p>
   493→                                <p className="text-[#F5C451] text-xs font-black mt-0.5">
   494→                                  {formatNaira(item.price)}
   495→                                </p>
   496→                              </div>
   497→                              {/* Edit Button */}
   498→                              <button
   499→                                onClick={() => startEdit(item)}
   500→                                disabled={isBusy}
   501→                                className="w-8 h-8 rounded-lg bg-white/5 border border-white/10 flex items-center justify-center shrink-0 hover:bg-white/10 transition-all disabled:opacity-50"
   502→                                aria-label="Edit"
   503→                              >
   504→                                <Pencil className="w-3.5 h-3.5 text-white/40" />
   505→                              </button>
   506→                            </div>
   507→
   508→                            {/* Item Stats */}
   509→                            <div className="flex items-center gap-3 mt-2">
   510→                              <div className="flex items-center gap-1">
   511→                                <ShoppingBag className="w-3 h-3 text-white/30" />
   512→                                <span className="text-white/30 text-[10px] font-semibold">
   513→                                  {item.reviewCount} reviews
   514→                                </span>
   515→                              </div>
   516→                              <div className="flex items-center gap-1">
   517→                                <TrendingUp className="w-3 h-3 text-[#10E07A]/40" />
