   100→
   101→function initials(name: string): string {
   102→  const parts = name.trim().split(/\s+/);
   103→  if (parts.length === 0) return '?';
   104→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   105→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   106→}
   107→
   108→function formatTime(ts: number): string {
   109→  const d = new Date(ts);
   110→  return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
   111→}
   112→
   113→/** Map an Order.status (DB string) → internal DeliveryStatus */
   114→function mapOrderStatus(status: string): DeliveryStatus {
   115→  switch (status) {
   116→    case 'Confirmed':
   117→    case 'Preparing':
   118→      return 'preparing';
   119→    case 'Ready':
   120→      return 'picked_up';
   121→    case 'In Transit':
   122→      return 'on_the_way';
   123→    case 'Delivered':
   124→      return 'delivered';
   125→    default:
   126→      return 'preparing';
   127→  }
   128→}
   129→
   130→/** Map Order.progress (0-100) → ETA minutes (inverse) */
   131→function progressToEta(progress: number, status: DeliveryStatus): number {
   132→  if (status === 'delivered') return 0;
   133→  if (status === 'arriving') return 2;
   134→  if (status === 'on_the_way') return Math.max(3, Math.round((100 - progress) / 6));
   135→  if (status === 'picked_up') return Math.max(5, Math.round((100 - progress) / 6));
   136→  return Math.max(8, Math.round((100 - progress) / 6));
   137→}
   138→
   139→/** Generate system messages based on status */
   140→function statusMessage(status: DeliveryStatus, riderName: string): string {
   141→  switch (status) {
   142→    case 'preparing':
   143→      return 'Your order is being prepared with care 🍳';
   144→    case 'picked_up':
   145→      return `${riderName} has picked up your order 🏍️`;
   146→    case 'on_the_way':
   147→      return `${riderName} is on the way to you 📍`;
   148→    case 'arriving':
   149→      return `${riderName} is arriving at your location 🚪`;
   150→    case 'delivered':
   151→      return `Delivered! Ramadan Mubarak 🌙 — Enjoy your meal!`;
   152→    default:
   153→      return 'Updating...';
   154→  }
   155→}

   160→
   161→export default function RealTimeTrackingModal() {
   162→  const { activeModal, setActiveModal, orders, userEmail } = useAppStore();
   163→  const { toast } = useToast();
   164→  const isOpen = activeModal === 'live-tracking';
   165→
   166→  const [delivery, setDelivery] = useState<DeliveryState | null>(null);
   167→  const [messages, setMessages] = useState<UpdateMessage[]>([]);
   168→  const [delivered, setDelivered] = useState(false);
   169→
   170→  // Derive the tracked order id from the store (no setState-in-effect needed)
   171→  const trackedOrderId =
   172→    orders.find((o) => o.status !== 'Delivered')?.id ??
   173→    orders[0]?.id ??
   174→    null;
   175→  // Use a ref so the polling callback always has the latest id without re-creating it
   176→  const trackedOrderIdRef = useRef<string | null>(trackedOrderId);
   177→  useEffect(() => {
   178→    trackedOrderIdRef.current = trackedOrderId;
   179→  }, [trackedOrderId]);
   180→
   181→  // ─── Socket.io for real-time updates ───
   182→  // Only join the room while the modal is open; the useSocket hook
   183→  // handles joining / leaving automatically.
   184→  const roomId = isOpen && trackedOrderId ? `order-${trackedOrderId}` : undefined;
   185→  const { socket, isConnected: socketConnected } = useSocket(roomId);
   186→
   187→  // Listen for order-status-update events
   188→  useEffect(() => {
   189→    if (!socket) return;
   190→    const onStatusUpdate = (payload: {
   191→      orderId?: string;
   192→      status?: string;
   193→      progress?: number;
   194→      riderName?: string;
   195→      eta?: number;
   196→      timestamp?: string;
   197→    }) => {
   198→      if (!payload || !payload.orderId) return;
   199→      // Only react to updates for the order we're currently tracking
   200→      if (
   201→        trackedOrderIdRef.current &&
   202→        payload.orderId !== trackedOrderIdRef.current
   203→      ) {
   204→        return;
   205→      }
   206→      const status = mapOrderStatus(payload.status || 'Preparing');
   207→      const progress =
   208→        typeof payload.progress === 'number' ? payload.progress : 0;
   209→      const eta =
   210→        typeof payload.eta === 'number'
   211→          ? payload.eta
   212→          : progressToEta(progress, status);
   213→      const riderName = payload.riderName || 'Your Rider';
   214→
   215→      setDelivery((prev) => {
   216→        // Preserve existing items / store / customer if present
   217→        const baseLat = 6.4541;
   218→        const baseLng = 3.3947;
   219→        const storeLat = baseLat + 0.01;
   220→        const storeLng = baseLng - 0.012;
   221→        const custLat = baseLat - 0.008;
   222→        const custLng = baseLng + 0.014;
   223→        const prevLoc = prev?.location;
   224→        const t = Math.min(1, Math.max(0, progress / 100));
   225→        const locLat =
   226→          prevLoc?.lat ?? storeLat + (custLat - storeLat) * t;
   227→        const locLng =
   228→          prevLoc?.lng ?? storeLng + (custLng - storeLng) * t;
   229→
   230→        const next: DeliveryState = {
   231→          orderId: payload.orderId,
   232→          rider: prev?.rider ?? {
   233→            name: riderName,
   234→            phone: '+234 800 000 0000',
   235→            photo: '',
   236→            rating: 4.9,
   237→            vehicle: 'Motorcycle',
   238→            color: '#38BDF8',
   239→          },
   240→          location: { lat: locLat, lng: locLng },
   241→          status,
   242→          eta,
   243→          progress,
   244→          customer: prev?.customer ?? { lat: custLat, lng: custLng },
   245→          store:
   246→            prev?.store ?? {
   247→              name: 'SwiftRamadan Kitchen',
   248→              lat: storeLat,
   249→              lng: storeLng,
   250→            },
   251→          total: prev?.total ?? 0,
   252→          items: prev?.items ?? [],
   253→        };
   254→
   255→        // Update rider name if changed
   256→        if (
   257→          payload.riderName &&
   258→          payload.riderName !== next.rider.name
   259→        ) {
   260→          next.rider = { ...next.rider, name: payload.riderName };
   261→        }
   262→
   263→        // Push a system message if status changed
   264→        if (lastStatusRef.current !== status) {
   265→          const prevStatus = lastStatusRef.current;
   266→          lastStatusRef.current = status;
   267→          if (prevStatus !== null) {
   268→            const msg: UpdateMessage = {
   269→              orderId: payload.orderId,
   270→              from: 'system',
   271→              text: statusMessage(status, next.rider.name),
   272→              timestamp: Date.now(),
   273→            };
   274→            setMessages((m) =>
   275→              m.some((x) => x.text === msg.text && x.from === 'system')
   276→                ? m
   277→                : [...m, msg]
   278→            );
   279→          }
   280→        }
   281→
   282→        return next;
   283→      });
   284→
   285→      if (status === 'delivered') {
   286→        setDelivered(true);
   287→      }
   288→    };
   289→
   290→    const onRiderLocation = (payload: {
   291→      orderId?: string;
   292→      lat?: number;
   293→      lng?: number;
   294→      progress?: number;
   295→    }) => {
   296→      if (!payload || !payload.orderId) return;
   297→      if (
   298→        trackedOrderIdRef.current &&
   299→        payload.orderId !== trackedOrderIdRef.current
   300→      ) {
   301→        return;
   302→      }
   303→      if (typeof payload.lat !== 'number' || typeof payload.lng !== 'number')
   304→        return;
   305→
   306→      setDelivery((prev) => {
   307→        if (!prev) return prev;
   308→        const next: DeliveryState = {
   309→          ...prev,
   310→          location: { lat: payload.lat!, lng: payload.lng! },
   311→          progress:
   312→            typeof payload.progress === 'number'
   313→              ? payload.progress
   314→              : prev.progress,
   315→        };
   316→        return next;
   317→      });
   318→    };
   319→
   320→    socket.on('order-status-update', onStatusUpdate);
   321→    socket.on('rider-location', onRiderLocation);
   322→    return () => {
   323→      socket.off('order-status-update', onStatusUpdate);
   324→      socket.off('rider-location', onRiderLocation);
   325→    };
   326→  }, [socket]);
   327→
   328→  // isPolling is purely derived from whether the modal is open AND
   329→  // the socket isn't connected (so we fall back to HTTP polling)
   330→  const isPolling = isOpen;
   331→
   332→  const messagesEndRef = useRef<HTMLDivElement>(null);
   333→  const chatScrollRef = useRef<HTMLDivElement>(null);
   334→  const lastStatusRef = useRef<DeliveryStatus | null>(null);
   335→
   336→  /* ─── Polling: fetch /api/orders every 3s when modal is open ─── */
   337→  const pollOrder = useCallback(async () => {
   338→    try {
   339→      const res = await fetch('/api/orders');
   340→      if (!res.ok) return;
   341→      const json = await res.json();
   342→      const allOrders: Array<{
   343→        id: string;
   344→        status: string;
   345→        total: number;
   346→        riderName: string | null;
   347→        items: Array<{ name: string; qty: number; price: number }>;
   348→        progress: number;
   349→        createdAt: string;
   350→      }> = json.orders ?? [];
   351→
   352→      const match =
   353→        allOrders.find((o) => o.id === trackedOrderIdRef.current) ??
   354→        allOrders.find((o) => o.status !== 'Delivered') ??
   355→        allOrders[0];
   356→
   357→      if (!match) return;
   358→
   359→      const status = mapOrderStatus(match.status);
   360→      const progress = match.progress ?? 0;
   361→      const eta = progressToEta(progress, status);
   362→
   363→      // Derive rider info — use Order.riderName if present, else placeholder
   364→      const riderName = match.riderName || 'Your Rider';
   365→      const rider: Rider = {
   366→        name: riderName,
   367→        phone: '+234 800 000 0000',
   368→        photo: '',
   369→        rating: 4.9,
   370→        vehicle: 'Motorcycle',
   371→        color: '#38BDF8',
   372→      };
   373→
   374→      // Stylized Lagos coordinates around the user's area
   375→      const baseLat = 6.4541;
   376→      const baseLng = 3.3947;
   377→
   378→      // Position the rider along a path from store → customer based on progress
   379→      const storeLat = baseLat + 0.01;
   380→      const storeLng = baseLng - 0.012;
   381→      const custLat = baseLat - 0.008;
   382→      const custLng = baseLng + 0.014;
   383→      const t = Math.min(1, Math.max(0, progress / 100));
   384→      const locLat = storeLat + (custLat - storeLat) * t;
   385→      const locLng = storeLng + (custLng - storeLng) * t;
   386→
   387→      const next: DeliveryState = {
   388→        orderId: match.id,
   389→        rider,
   390→        location: { lat: locLat, lng: locLng },
   391→        status,
   392→        eta,
   393→        progress,
   394→        customer: { lat: custLat, lng: custLng },
   395→        store: { name: 'SwiftRamadan Kitchen', lat: storeLat, lng: storeLng },
   396→        total: match.total,
   397→        items: match.items ?? [],
   398→      };
   399→
   400→      setDelivery((prev) => {
   401→        // If status changed, push a system message
   402→        if (lastStatusRef.current !== status) {
   403→          const prevStatus = lastStatusRef.current;
   404→          lastStatusRef.current = status;
   405→          if (prevStatus !== null) {
   406→            const msg: UpdateMessage = {
   407→              orderId: match.id,
   408→              from: 'system',
   409→              text: statusMessage(status, riderName),
   410→              timestamp: Date.now(),
   411→            };
   412→            setMessages((m) =>
   413→              m.some((x) => x.text === msg.text && x.from === 'system')
   414→                ? m
   415→                : [...m, msg]
   416→            );
   417→          }
   418→        }
   419→        return next;
   420→      });
   421→
   422→      // Trigger delivered state
   423→      if (status === 'delivered') {
   424→        setDelivered(true);
   425→      }
   426→    } catch (err) {
   427→      console.error('Polling error:', err);
   428→    }
   429→  }, []);
   430→
   431→  useEffect(() => {
   432→    if (!isOpen) return;
   433→    // Initial fetch immediately
   434→    pollOrder();
   435→    // Fall back to polling every 5s. If the socket is connected, the
   436→    // realtime service pushes updates instantly and we don't need to
   437→    // poll as aggressively — but we still poll every 5s as a safety net
   438→    // in case the socket disconnects silently.
   439→    const interval = setInterval(pollOrder, 5000);
   440→    return () => {
   441→      clearInterval(interval);
   442→      // Don't reset delivery here so the close animation doesn't flash
   443→    };
   444→  }, [isOpen, pollOrder]);
   445→
   446→  /* Auto-scroll updates to bottom */
   447→  useEffect(() => {
   448→    if (chatScrollRef.current) {
   449→      chatScrollRef.current.scrollTop = chatScrollRef.current.scrollHeight;
   450→    }
   451→  }, [messages]);
   452→
   453→  /* ─── handlers ─── */
   454→  const handleClose = useCallback(() => {
   455→    setActiveModal(null);
   456→  }, [setActiveModal]);
   457→
   458→  const handleCallRider = useCallback(() => {
   459→    if (!delivery?.rider) return;
   460→    toast({
   461→      title: `Calling ${delivery.rider.name}... 📞`,
   462→      description: delivery.rider.phone,
   463→    });
   464→  }, [delivery, toast]);
   465→
   466→  /** Trigger the rate-delivery modal — stash order info in localStorage since
   467→   *  store.ts cannot be modified (RateDeliveryModal owner = Agent E). */
   468→  const handleRateRider = useCallback(() => {

   470→    try {
   471→      localStorage.setItem(
   472→        'rateDeliveryOrder',
   473→        JSON.stringify({
   474→          orderId: delivery.orderId,
   475→          riderName: delivery.rider.name,
   476→          total: delivery.total,
   477→          items: delivery.items,
   478→          userEmail,
   479→        })
   480→      );
   481→    } catch (e) {

   483→    }

   485→  }, [delivery, userEmail, setActiveModal]);
   486→
   487→  /* ─── derived ─── */
   488→  const currentStageIndex = useMemo(() => {
   489→    if (!delivery) return 0;
   490→    const idx = STATUS_ORDER.indexOf(delivery.status);
   491→    return idx < 0 ? 0 : idx;
   492→  }, [delivery]);
   493→
   494→  // Map coordinates → pixel positions on the map panel (percentage)
   495→  const mapLayout = useMemo(() => {
   496→    if (!delivery) return null;
   497→    const pts = [delivery.store, delivery.customer, delivery.location];
   498→    const lats = pts.map((p) => p.lat);
   499→    const lngs = pts.map((p) => p.lng);
   500→    const minLat = Math.min(...lats);
   501→    const maxLat = Math.max(...lats);
   502→    const minLng = Math.min(...lngs);
   503→    const maxLng = Math.max(...lngs);
   504→    const latRange = Math.max(maxLat - minLat, 0.001);
   505→    const lngRange = Math.max(maxLng - minLng, 0.001);
   506→    const pad = 0.15;
   507→    const project = (p: GeoPoint) => {
   508→      const x = ((p.lng - minLng) / lngRange) * (1 - pad * 2) + pad;
   509→      const y = 1 - (((p.lat - minLat) / latRange) * (1 - pad * 2) + pad);
   510→      return { x: x * 100, y: y * 100 };
   511→    };
   512→    const store = project(delivery.store);
   513→    const customer = project(delivery.customer);
   514→    const rider = project(delivery.location);
   515→    return { store, customer, rider };
   516→  }, [delivery]);
   517→
   518→  return (
   519→    <AnimatePresence>
   520→      {isOpen && (
   521→        <>
   522→          {/* Backdrop */}
   523→          <motion.div
   524→            initial={{ opacity: 0 }}
   525→            animate={{ opacity: 1 }}
   526→            exit={{ opacity: 0 }}
   527→            className="fixed inset-0 bg-black/80 z-[95]"
   528→            onClick={handleClose}
   529→          />
   530→
   531→          {/* Full-screen modal */}
   532→          <motion.div
   533→            initial={{ y: '100%' }}
   534→            animate={{ y: 0 }}
   535→            exit={{ y: '100%' }}
   536→            transition={{ type: 'spring', damping: 28, stiffness: 220 }}
   537→            className="fixed inset-0 z-[100] bg-[#0B0D14] flex flex-col overflow-hidden"
   538→          >
   539→            {/* ─── Sticky Header ─── */}
   540→            <div className="glass-effect border-b border-white/5 px-4 py-3 flex items-center justify-between shrink-0 z-20">
   541→              <div className="flex items-center gap-3 min-w-0">
   542→                <div className="w-10 h-10 rounded-xl bg-[#38BDF8]/10 border border-[#38BDF8]/30 flex items-center justify-center shrink-0">
   543→                  <MapPin className="w-5 h-5 text-[#38BDF8]" />
   544→                </div>
   545→                <div className="min-w-0">
   546→                  <h2 className="text-white text-base font-bold leading-tight flex items-center gap-2">
   547→                    Live Tracking
   548→                    {socketConnected ? (
   549→                      <motion.span
   550→                        animate={{ opacity: [1, 0.4, 1] }}
   551→                        transition={{ duration: 1.5, repeat: Infinity }}
   552→                        className="w-2 h-2 rounded-full bg-[#10E07A] inline-block"
   553→                        title="Realtime connected"
   554→                      />
   555→                    ) : isPolling ? (
   556→                      <motion.span
   557→                        animate={{ opacity: [1, 0.4, 1] }}
   558→                        transition={{ duration: 1.5, repeat: Infinity }}
   559→                        className="w-2 h-2 rounded-full bg-[#38BDF8] inline-block"
   560→                        title="Polling (socket offline)"
   561→                      />
   562→                    ) : (
   563→                      <span className="w-2 h-2 rounded-full bg-white/30 inline-block" />
   564→                    )}
   565→                  </h2>
   566→                  <p className="text-white/40 text-[11px] truncate">
   567→                    {delivery
   568→                      ? `Order #${delivery.orderId.slice(-6).toUpperCase()} • ${delivery.rider.name}`
   569→                      : 'Connecting to rider...'}
   570→                  </p>
   571→                </div>
   572→              </div>
   573→              <button
   574→                onClick={handleClose}
   575→                className="w-10 h-10 flex items-center justify-center rounded-full bg-white/5 hover:bg-white/10 transition-colors shrink-0"
   576→                aria-label="Close live tracking"
   577→              >
   578→                <X className="w-5 h-5 text-white/60" />
   579→              </button>
   580→            </div>
   581→
   582→            {/* ─── Scrollable content ─── */}
   583→            <div className="flex-1 overflow-y-auto custom-scrollbar">
   584→              {/* Map panel */}
   585→              <MapPanel delivery={delivery} layout={mapLayout} />
   586→
   587→              {/* Delivered banner */}
   588→              <AnimatePresence>
   589→                {delivered && (
   590→                  <motion.section
   591→                    initial={{ opacity: 0, y: -10 }}
   592→                    animate={{ opacity: 1, y: 0 }}
   593→                    exit={{ opacity: 0 }}
