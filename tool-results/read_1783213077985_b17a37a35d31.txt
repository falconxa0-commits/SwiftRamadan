    35→function ConfettiParticle({ delay, color }: { delay: number; color: string }) {
    36→  const randomX = Math.random() * 300 - 150;
    37→  const randomY = Math.random() * 400 + 100;
    38→  const randomRotate = Math.random() * 720 - 360;
    39→  const size = Math.random() * 8 + 4;
    40→
    41→  return (
    42→    <motion.div
    43→      initial={{ opacity: 1, x: 0, y: 0, rotate: 0, scale: 1 }}
    44→      animate={{ opacity: 0, x: randomX, y: randomY, rotate: randomRotate, scale: 0.3 }}
    45→      transition={{ duration: 1.8, delay, ease: 'easeOut' }}
    46→      className="absolute rounded-sm pointer-events-none"
    47→      style={{
    48→        width: size,
    49→        height: size,
    50→        backgroundColor: color,
    51→        left: '50%',
    52→        top: '30%',
    53→      }}
    54→    />
    55→  );
    56→}
