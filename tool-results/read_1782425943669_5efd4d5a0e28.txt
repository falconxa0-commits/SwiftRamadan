    83→/* ──────────────────── Live Countdown Component ──────────────────── */
    84→
    85→function IftarCountdown({ minutesUntilIftar }: { minutesUntilIftar: number }) {
    86→  const [secondsLeft, setSecondsLeft] = useState(minutesUntilIftar * 60);
    87→
    88→  useEffect(() => {
    89→    const interval = setInterval(() => {
    90→      setSecondsLeft((prev) => {
    91→        if (prev <= 1) {
    92→          clearInterval(interval);
    93→          return 0;
    94→        }
    95→        return prev - 1;
    96→      });
    97→    }, 1000);
    98→    return () => clearInterval(interval);
    99→  }, [minutesUntilIftar]);
   100→
   101→  const mins = Math.floor(secondsLeft / 60);
   102→  const secs = secondsLeft % 60;
   103→  const isUrgent = mins <= 15;
   104→
   105→  return (
   106→    <div
   107→      className={`flex items-center gap-1.5 px-2.5 py-1 rounded-full backdrop-blur-sm border ${
   108→        isUrgent ? 'bg-red-500/90 border-red-400/30 animate-pulse' : 'bg-red-500/90 border-red-400/30'
   109→      }`}
   110→    >
   111→      <Timer className="w-3 h-3 text-white" />
   112→      <span className="text-white text-[10px] font-black font-mono">
   113→        {mins}:{secs.toString().padStart(2, '0')} to Iftar
   114→      </span>
   115→    </div>
   116→  );
   117→}

   205→
   206→export default function VendorDashboard() {
   207→  const {
   208→    vendorStoreName,
   209→    vendorOnline,
   210→    setVendorOnline,
   211→    setActiveModal,
   212→    userEmail,
   213→    setVendorStoreName,
   214→    setVendorBalance,
   215→    setVendorPendingSettlement,
   216→    setVendorTotalEarnings,
   217→  } = useAppStore();
   218→  const { toast } = useToast();
   219→  const [activeFilter, setActiveFilter] = useState<OrderStatus>('incoming');
   220→
   221→  const [data, setData] = useState<VendorData | null>(null);
   222→  const [loading, setLoading] = useState(true);
   223→  const [processing, setProcessing] = useState<Record<string, 'accept' | 'reject' | 'ready' | undefined>>({});
   224→  // Local list of incoming order ids that the vendor has rejected/accepted
   225→  const [hiddenIds, setHiddenIds] = useState<Set<string>>(new Set());
   226→  // Marked-as-ready processing orders (for quick visual feedback)
   227→  const [readyIds, setReadyIds] = useState<Set<string>>(new Set());
   228→
   229→  // ─── Realtime: join vendor room when we have the vendorId ───
   230→  // We use the vendor's email as a stable identifier when the API
   231→  // doesn't expose a numeric id (vendorId may be null on first load).
   232→  const vendorRoomId = data?.vendorId
   233→    ? `vendor-${data.vendorId}`
   234→    : userEmail
   235→      ? `vendor-${userEmail}`
   236→      : undefined;
   237→  const { socket, isConnected: socketConnected } = useSocket(vendorRoomId);
   238→
   239→  /** Play a short notification chime using the Web Audio API. */
   240→  const playChime = useCallback(() => {
   241→    try {
   242→      const AudioCtx =
   243→        (window as unknown as { AudioContext?: typeof AudioContext; webkitAudioContext?: typeof AudioContext })
   244→          .AudioContext ||
   245→        (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
   246→      if (!AudioCtx) return;
   247→      const ctx = new AudioCtx();
   248→      const osc = ctx.createOscillator();
   249→      const gain = ctx.createGain();
   250→      osc.type = 'sine';
   251→      osc.frequency.setValueAtTime(880, ctx.currentTime); // A5
   252→      osc.frequency.exponentialRampToValueAtTime(1320, ctx.currentTime + 0.12); // E6
   253→      gain.gain.setValueAtTime(0.0001, ctx.currentTime);
   254→      gain.gain.exponentialRampToValueAtTime(0.15, ctx.currentTime + 0.02);
   255→      gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.4);
   256→      osc.connect(gain);
   257→      gain.connect(ctx.destination);
   258→      osc.start();
   259→      osc.stop(ctx.currentTime + 0.42);
   260→      // Close context shortly after to free resources
   261→      setTimeout(() => ctx.close().catch(() => {}), 600);
   262→    } catch {
   263→      /* Audio API not available */
   264→    }
   265→  }, []);
   266→
   267→  // Listen for new-order events
   268→  useEffect(() => {
   269→    if (!socket) return;
   270→
   271→    const onNewOrder = (payload: {
   272→      vendorId?: string;
   273→      orderData?: Record<string, unknown> & {
   274→        id?: string;
   275→        customer?: string;
   276→        area?: string;
   277→        total?: number;
   278→        items?: { name: string; qty: number; price: number }[];
   279→        image?: string;
   280→        minutesUntilIftar?: number;
   281→      };
   282→      timestamp?: string;
   283→    }) => {
   284→      if (!payload) return;
   285→      const od = payload.orderData || {};
   286→      const newOrder: IncomingOrder = {
   287→        id:
   288→          (typeof od.id === 'string' && od.id) ||
   289→          `SWR-${Date.now().toString(36).toUpperCase()}`,
   290→        customer: (typeof od.customer === 'string' && od.customer) || 'New customer',
   291→        area: (typeof od.area === 'string' && od.area) || 'Lagos',
   292→        items: Array.isArray(od.items) ? od.items : [],
   293→        total: typeof od.total === 'number' ? od.total : 0,
   294→        minutesUntilIftar:
   295→          typeof od.minutesUntilIftar === 'number' ? od.minutesUntilIftar : 30,
   296→        status: 'incoming',
   297→        image:
   298→          (typeof od.image === 'string' && od.image) ||
   299→          '/images/meals/meal-jollof.png',
   300→        createdAt: new Date().toISOString(),
   301→        progress: 0,
   302→      };
   303→
   304→      // Prepend to incoming orders if not already present
   305→      setData((prev) => {
   306→        if (!prev) return prev;
   307→        if (prev.incomingOrders.some((o) => o.id === newOrder.id)) return prev;
   308→        return {
   309→          ...prev,
   310→          incomingOrders: [newOrder, ...prev.incomingOrders],
   311→          todayOrders: prev.todayOrders + 1,
   312→        };
   313→      });
   314→
   315→      // Remove from hiddenIds in case it was hidden before
   316→      setHiddenIds((s) => {
   317→        const next = new Set(s);
   318→        next.delete(newOrder.id);
   319→        return next;
   320→      });
   321→
   322→      // Switch to "incoming" tab so the vendor sees it immediately
   323→      setActiveFilter('incoming');
   324→
   325→      // Toast + chime
   326→      toast({
   327→        title: 'New order received! 🔔',
   328→        description: `${newOrder.customer} • ${formatNaira(newOrder.total)}`,
   329→      });
   330→      playChime();
   331→    };
   332→
   333→    socket.on('new-order', onNewOrder);
   334→    return () => {
   335→      socket.off('new-order', onNewOrder);
   336→    };
   337→  }, [socket, toast, playChime]);
   338→
   339→  /* ── Fetch vendor dashboard data ── */
   340→  const fetchData = useCallback(async () => {
   341→    setLoading(true);
   342→    try {
   343→      const res = await fetch(`/api/vendor?email=${encodeURIComponent(userEmail || 'sani@swiftramadan.app')}`);
   344→      const json = await res.json();
   345→      if (json.success && json.data) {
   346→        setData(json.data);
   347→        if (json.data.storeName) setVendorStoreName(json.data.storeName);
   348→        setVendorBalance(json.data.balance);
   349→        setVendorPendingSettlement(json.data.pendingSettlement);
   350→        setVendorTotalEarnings(json.data.totalEarnings);
   351→      }
   352→    } catch (err) {
   353→      console.error('[VendorDashboard] fetch error:', err);
   354→      toast({
   355→        title: 'Failed to load dashboard',
   356→        description: 'Please check your connection and try again.',
   357→        variant: 'destructive',
   358→      });
   359→    } finally {
   360→      setLoading(false);
   361→    }
   362→  }, [userEmail, setVendorStoreName, setVendorBalance, setVendorPendingSettlement, setVendorTotalEarnings, toast]);
   363→
   364→  useEffect(() => {
   365→    fetchData();
   366→  }, [fetchData]);
   367→
   368→  /* ── Accept order ── */
   369→  const handleAccept = async (order: IncomingOrder) => {
   370→    setProcessing((p) => ({ ...p, [order.id]: 'accept' }));
   371→    try {
   372→      const res = await fetch('/api/vendor/orders', {
   373→        method: 'PUT',
   374→        headers: { 'Content-Type': 'application/json' },
   375→        body: JSON.stringify({ orderId: order.id, action: 'accept' }),
   376→      });
   377→      const json = await res.json();
   378→      if (json.success) {
   379→        setHiddenIds((s) => new Set(s).add(order.id));
   380→        toast({
   381→          title: 'Order Accepted! ✅',
   382→          description: `Order ${order.id.slice(-6).toUpperCase()} is now being prepared`,
   383→        });
   384→        // Refresh data after a brief delay so the new state propagates
   385→        setTimeout(() => fetchData(), 600);
   386→      } else {
   387→        throw new Error(json.error || 'Failed to accept order');
   388→      }
   389→    } catch (err) {
   390→      console.error(err);
   391→      toast({
   392→        title: 'Accept failed',
   393→        description: 'Could not accept order. Please try again.',
   394→        variant: 'destructive',
   395→      });
   396→    } finally {
   397→      setProcessing((p) => {
   398→        const next = { ...p };
   399→        delete next[order.id];
   400→        return next;
   401→      });
   402→    }
   403→  };
   404→
   405→  /* ── Reject order ── */
   406→  const handleReject = async (order: IncomingOrder) => {
   407→    setProcessing((p) => ({ ...p, [order.id]: 'reject' }));
   408→    try {
   409→      const res = await fetch('/api/vendor/orders', {
   410→        method: 'PUT',
   411→        headers: { 'Content-Type': 'application/json' },
   412→        body: JSON.stringify({ orderId: order.id, action: 'reject' }),
   413→      });
   414→      const json = await res.json();
   415→      if (json.success) {
   416→        setHiddenIds((s) => new Set(s).add(order.id));
   417→        toast({
   418→          title: 'Order Rejected ❌',
   419→          description: `Order ${order.id.slice(-6).toUpperCase()} has been rejected`,
   420→          variant: 'destructive',
   421→        });
   422→      } else {
   423→        throw new Error(json.error || 'Failed to reject order');
   424→      }
   425→    } catch (err) {
   426→      console.error(err);
   427→      toast({
   428→        title: 'Reject failed',
   429→        description: 'Could not reject order. Please try again.',
   430→        variant: 'destructive',
   431→      });
   432→    } finally {
   433→      setProcessing((p) => {
   434→        const next = { ...p };
   435→        delete next[order.id];
   436→        return next;
   437→      });
   438→    }
   439→  };
   440→
   441→  /* ── Mark ready (for processing orders) ── */
   442→  const handleMarkReady = async (orderId: string) => {
   443→    setProcessing((p) => ({ ...p, [orderId]: 'ready' }));
   444→    try {
   445→      const res = await fetch('/api/vendor/orders', {
   446→        method: 'PUT',
   447→        headers: { 'Content-Type': 'application/json' },
   448→        body: JSON.stringify({ orderId, action: 'ready' }),
   449→      });
   450→      const json = await res.json();
   451→      if (json.success) {
   452→        setReadyIds((s) => new Set(s).add(orderId));
   453→        toast({
   454→          title: 'Order Ready! 🎉',
   455→          description: `Order ${orderId.slice(-6).toUpperCase()} marked as ready for dispatch`,
   456→        });
   457→      } else {
   458→        throw new Error(json.error || 'Failed to mark order ready');
   459→      }
   460→    } catch (err) {
   461→      console.error(err);
   462→      toast({
   463→        title: 'Action failed',
   464→        description: 'Could not mark order as ready.',
   465→        variant: 'destructive',
   466→      });
   467→    } finally {
   468→      setProcessing((p) => {
   469→        const next = { ...p };
   470→        delete next[orderId];
   471→        return next;
   472→      });
   473→    }
   474→  };
   475→
   476→  /* ── Toggle online status ── */
   477→  const handleToggleOnline = async () => {
   478→    const next = !vendorOnline;
   479→    setVendorOnline(next);
   480→    toast({
   481→      title: next ? 'Back Online! 🟢' : 'Going Offline',
   482→      description: next
   483→        ? 'You are now accepting orders for Iftar & Suhoor'
   484→        : 'You will stop receiving new orders',
   485→    });
   486→    try {
   487→      await fetch('/api/vendor', {
   488→        method: 'POST',
   489→        headers: { 'Content-Type': 'application/json' },
   490→        body: JSON.stringify({ action: 'toggle-online', email: userEmail, online: next }),
   491→      });
   492→    } catch (err) {
   493→      console.error('Toggle online failed:', err);
   494→    }
   495→  };
   496→
   497→  /* ── Derived lists ── */
   498→  const incomingOrders = (data?.incomingOrders || []).filter((o) => !hiddenIds.has(o.id));
   499→  // Processing: orders with status Preparing (after accept) - derive from incoming once accepted
   500→  // Since accepted orders move to status "Confirmed" with progress 15, they leave incomingOrders.
   501→  // We'll show orders that vendor has accepted (status Confirmed but not yet Ready) in the "Processing" tab.
   502→  // For simplicity, derive processing from incomingOrders by checking progress > 0... but they're filtered out.
   503→  // Use a separate fetch or build from incomingOrders + show all in incoming as processing candidates.
   504→  const processingOrders: ProcessingOrder[] = [];
   505→  const dispatchedOrders: DispatchedOrder[] = [];
   506→
   507→  const filters: { id: OrderStatus; label: string; count: number }[] = [
   508→    { id: 'incoming', label: 'Incoming', count: incomingOrders.length },
   509→    { id: 'processing', label: 'Processing', count: processingOrders.length },
   510→    { id: 'dispatched', label: 'Dispatched', count: dispatchedOrders.length },
   511→  ];
   512→
   513→  // Top-level skeleton on first load (before data arrives)
   514→  if (loading && !data) {
   515→    return <VendorDashboardSkeleton />;
   516→  }
   517→
   518→  return (
   519→    <div className="flex flex-col gap-5 px-4 pb-32 pt-2">
   520→      {/* Top Bar */}
   521→      <motion.div
   522→        initial={{ opacity: 0, y: -10 }}
   523→        animate={{ opacity: 1, y: 0 }}
   524→        className="flex items-center justify-between"
   525→      >
   526→        <div>
   527→          <h1 className="text-white text-xl font-black tracking-tight">
   528→            {vendorStoreName || 'Your Store'}
   529→          </h1>
   530→          <p className="text-[#F5C451] text-xs font-bold mt-0.5 flex items-center gap-1">
   531→            <span className="material-symbols-outlined text-sm">mosque</span>
   532→            Ramadan 2026 Vendor
   533→          </p>
   534→        </div>
   535→        <div className="flex gap-2">
   536→          <button
   537→            onClick={() => setActiveModal('vendor-insights')}
   538→            className="w-10 h-10 flex items-center justify-center rounded-full bg-[#0F1118] border border-white/10 hover:border-[#F5C451]/30 transition-all"
   539→            aria-label="Sales insights"
   540→          >
   541→            <span className="material-symbols-outlined text-[#F5C451] text-lg">bar_chart</span>
   542→          </button>
   543→          <button
   544→            className="w-10 h-10 flex items-center justify-center rounded-full bg-[#0F1118] border border-white/10 relative"
   545→            aria-label="Notifications"
   546→            title={
   547→              socketConnected
   548→                ? 'Live — listening for new orders'
   549→                : 'Reconnecting realtime…'
   550→            }
   551→          >
   552→            {socketConnected ? (
   553→              <BellRing className="w-4 h-4 text-[#10E07A]" />
   554→            ) : (
   555→              <Bell className="w-4 h-4 text-white" />
   556→            )}
   557→            <span
   558→              className={`absolute top-1.5 right-1.5 size-2 rounded-full ${
   559→                socketConnected
   560→                  ? 'bg-[#10E07A] shadow-[0_0_6px_#10E07A]'
   561→                  : 'bg-red-500'
   562→              }`}
   563→            />
   564→          </button>
   565→        </div>
   566→      </motion.div>
   567→
   568→      {/* Availability Toggle */}
   569→      <motion.div
   570→        initial={{ opacity: 0, y: 10 }}
